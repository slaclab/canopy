1"""Request-input parsing shared by the module routers. 

2 

3Primary keys are ``UUID(as_uuid=False)`` columns, and with asyncpg's native UUID support 

4SQLAlchemy installs no bind processor for them: whatever string a router passes on reaches 

5asyncpg unchanged. asyncpg encodes only the canonical and bare-hex spellings, so an id or 

6continuation token in any other form raises ``DataError`` inside the query and surfaces as 

7a 500. Routers parse ids here and pass on the canonical form, turning anything else into 

8their own 404 or 400 before a query is built. 

9""" 

10 

11from uuid import UUID 

12 

13from pydantic import TypeAdapter, ValidationError 

14 

15from canopy.core.errors import BadRequest, CanopyError 

16 

17# pydantic's parser rather than uuid.UUID(): the latter ends in int(hex, 16), which accepts a 

18# leading sign, whitespace and underscores, so "+" followed by 31 zeros would parse as the nil 

19# UUID instead of being rejected. 

20_UUID = TypeAdapter(UUID) 

21 

22 

23def parse_uuid(value: str) -> str | None: 

24 """Canonical lowercase, hyphenated form of ``value``, or ``None`` when it is not a UUID. 

25 

26 Braced, ``urn:uuid:``, bare-hex and uppercase spellings are accepted and normalised. 

27 """ 

28 try: 

29 return str(_UUID.validate_python(value)) 

30 except ValidationError: 

31 return None 

32 

33 

34def require_uuid(value: str, error: type[CanopyError], message: str) -> str: 

35 """``parse_uuid`` that raises ``error(message)`` instead of returning ``None``. 

36 

37 Routers use it on path ids so a malformed id becomes the same 404 a missing row would. 

38 """ 

39 canonical = parse_uuid(value) 

40 if canonical is None: 

41 raise error(message) 

42 return canonical 

43 

44 

45def parse_continuation_token(value: str | None) -> str | None: 

46 """Canonical form of a UUID keyset ``continuationToken``, or ``None`` for the first page. 

47 

48 An absent or empty token means the first page. Anything else that is not a UUID raises 

49 ``BadRequest`` so the caller answers 400 rather than failing inside asyncpg. 

50 """ 

51 if not value: 

52 return None 

53 return require_uuid(value, BadRequest, "Invalid continuationToken")
