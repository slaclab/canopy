1"""Live data ingestion from Hare's PVBroker into Vole ring buffers. 

2 

3Vole registers as an internal "client" of the Hare broker using the same 

4add_client/subscribe API that WebSocket clients use. Diff frames arrive 

5at ~100ms intervals; each PVValue is converted to a Sample and appended 

6to the corresponding ring. 

7 

8This module is guarded by try/except at import sites — it degrades 

9gracefully when Hare is not available (tests, standalone Mole-only mode). 

10""" 

11

15from typing import Any

18from canopy.epics.pv_protocol import strip_protocol_prefix 

19from canopy.vole.services.buffer_service import BufferService

24class VoleChannel: 

25 """Implements Hare's ClientChannel protocol for in-process ingestion. 

26 

27 The broker calls ``send_json`` with diff/initial/heartbeat frames. 

28 Vole only cares about "diff" and "initial" frames. 

29 """ 

30 

31 def __init__(self, ingest: HareIngestService) -> None: 

32 self._ingest = ingest 

33 

34 async def send_json(self, data: Any) -> None: 

35 # Never let an ingestion error escape: the broker treats a send_json 

36 # exception as a dead websocket and would silently unregister Vole, 

37 # permanently stopping live ingestion for every PV. A malformed frame 

38 # must cost us at most that frame. 

39 try: 

40 if not isinstance(data, dict): 

41 return 

42 frame_type = data.get("type") 

43 if frame_type in ("diff", "initial"): 

44 pv_data = data.get("data") 

45 if isinstance(pv_data, dict): 

46 await self._ingest._handle_frame(pv_data) 

47 except Exception: 

48 logger.warning( 

49 "Failed to ingest %r frame", 

50 data.get("type") if isinstance(data, dict) else type(data).__name__, 

51 exc_info=True, 

52 ) 

53 

54 async def close(self) -> None: 

55 pass

58class HareIngestService: 

59 """Bridges Hare's PVBroker to Vole's BufferService for live data ingestion.""" 

60 

61 CLIENT_ID = "vole-internal" 

62 

63 def __init__(self, buffer_service: BufferService, broker: Any) -> None: 

64 self._buffer = buffer_service 

65 self._broker = broker 

66 self._channel = VoleChannel(self) 

67 self._subscribed_pvs: set[str] = set() 

68 self._running = False 

69 

70 async def start(self) -> None: 

71 self._broker.add_client(self.CLIENT_ID, self._channel) 

72 self._running = True 

73 logger.info("HareIngestService started (client_id=%s)", self.CLIENT_ID) 

74 

75 async def stop(self) -> None: 

76 if not self._running: 

77 return 

78 self._running = False 

79 if self._subscribed_pvs: 

80 await self._broker.unsubscribe(self.CLIENT_ID, list(self._subscribed_pvs)) 

81 self._subscribed_pvs.clear() 

82 await self._broker.remove_client(self.CLIENT_ID) 

83 logger.info("HareIngestService stopped") 

84 

85 async def subscribe(self, pvs: list[str]) -> list[str]: 

86 """Subscribe Vole to live updates for these PVs via Hare. 

87 

88 Returns which of ``pvs`` are subscribed afterwards. Only names the broker 

89 accepted are recorded — a rejected one would make GraceManager believe 

90 there is a buffer to tear down. 

91 """ 

92 new_pvs = [pv for pv in pvs if pv not in self._subscribed_pvs] 

93 if new_pvs: 

94 accepted = await self._broker.subscribe(self.CLIENT_ID, new_pvs) 

95 self._subscribed_pvs.update(accepted) 

96 logger.debug("Vole subscribed to %d/%d new PVs: %s", len(accepted), len(new_pvs), accepted[:5]) 

97 return [pv for pv in pvs if pv in self._subscribed_pvs] 

98 

99 async def unsubscribe(self, pvs: list[str]) -> None: 

100 """Unsubscribe Vole from live updates for these PVs.""" 

101 to_unsub = [pv for pv in pvs if pv in self._subscribed_pvs] 

102 if not to_unsub: 

103 return 

104 await self._broker.unsubscribe(self.CLIENT_ID, to_unsub) 

105 self._subscribed_pvs -= set(to_unsub) 

106 logger.debug("Vole unsubscribed from %d PVs", len(to_unsub)) 

107 

108 @property 

109 def subscribed_pvs(self) -> set[str]: 

110 return set(self._subscribed_pvs) 

111 

112 async def _handle_frame(self, data: dict[str, dict]) -> None: 

113 """Process a diff/initial frame: convert each PVValue to a Sample and append. 

114 

115 The whole per-PV body is guarded so one malformed PV entry (e.g. a 

116 non-numeric ``tsNs``) skips only that PV — the rest of the frame still 

117 lands, and no exception propagates back to the broker. 

118 """ 

119 for pv_name, pv_dict in data.items(): 

120 if not isinstance(pv_dict, dict): 

121 continue 

122 if not pv_dict.get("connected", True): 

123 continue 

124 

125 try: 

126 sample = self._pv_value_to_sample(pv_dict) 

127 if sample is None: 

128 continue 

129 canonical_pv = strip_protocol_prefix(pv_name) 

130 await self._buffer.append(canonical_pv, sample) 

131 except Exception: 

132 logger.warning("Failed to ingest live sample for %s", pv_name, exc_info=True) 

133 

134 @staticmethod 

135 def _pv_value_to_sample(pv_dict: dict) -> Sample | None: 

136 """Convert a PVValue wire dict to a Sample for ring storage.""" 

137 ts_ns = pv_dict.get("tsNs") or pv_dict.get("ts_ns") 

138 value = pv_dict.get("value") 

139 severity_raw = pv_dict.get("severity", "NO_ALARM") 

140 

141 if ts_ns is None: 

142 return None 

143 # A bool is an int in Python but never a valid timestamp; anything 

144 # non-numeric (string, dict, ...) would blow up in int() below. 

145 if isinstance(ts_ns, bool) or not isinstance(ts_ns, int | float): 

146 logger.debug("Skipping sample for %s: non-numeric tsNs %r", pv_dict.get("pv"), ts_ns) 

147 return None 

148 

149 if isinstance(severity_raw, str): 

150 try: 

151 severity = AlarmSeverity(severity_raw) 

152 except ValueError: 

153 severity = AlarmSeverity.INVALID 

154 else: 

155 severity = AlarmSeverity.INVALID 

156 

157 return Sample( 

158 ts_ns=int(ts_ns), 

159 value=value, 

160 severity=severity, 

161 source="live", 

162 )
