13def _float(value: object) -> float | np.ndarray: 

14 """Convert a scalar or array value to floating point.""" 

15 if isinstance(value, np.ndarray): 

16 if np.ma.isMaskedArray(value): 

17 return value.astype(np.float64).filled(np.nan) 

18 return np.asarray(value, dtype=np.float64) 

19 return float(value)

22def _int(value: object) -> int | np.ndarray: 

23 """Convert a scalar or array to a signed 64-bit integer. 

24 

25 Array NaN and infinity values remain NaN; finite values are truncated 

26 toward zero and returned as float64 so missing samples stay missing. 

27 """ 

28 if not isinstance(value, np.ndarray): 

29 return int(value) 

30 

31 if np.issubdtype(value.dtype, np.integer) or np.issubdtype(value.dtype, np.bool_): 

32 data = np.ma.getdata(value) 

33 integers = data[~np.ma.getmaskarray(value)].astype(object) 

34 if np.any(integers < -(2**63)) or np.any(integers >= 2**63): 

35 raise OverflowError("int() array values must fit in signed 64-bit integers") 

36 return value.astype(np.int64) 

37 

38 if np.ma.isMaskedArray(value): 

39 value = value.astype(np.float64).filled(np.nan) 

40 

41 values = np.asarray(value, dtype=np.float64) 

42 finite = np.isfinite(values) 

43 finite_values = np.trunc(values[finite]) 

44 if np.any(finite_values < -(2**63)) or np.any(finite_values >= 2**63): 

45 raise OverflowError("int() array values must fit in signed 64-bit integers") 

46 result = np.full(values.shape, np.nan, dtype=np.float64) 

47 result[finite] = finite_values 

48 return result

51def _str(value: object) -> str: 

52 """Convert a scalar value to text; array conversion is unsupported.""" 

53 if isinstance(value, np.ndarray): 

54 raise TypeError("str() is only supported for scalar values")

58def _isnan(value: object) -> bool | np.ndarray: 

59 """Test scalar or array values for NaN.""" 

60 if isinstance(value, np.ndarray): 

61 if np.ma.isMaskedArray(value): 

62 return np.ma.filled(np.isnan(value), True) 

63 return np.isnan(value) 

64 return bool(np.isnan(value))
