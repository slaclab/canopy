73def _import_sub_app(name: str, package: str): 

74 """Import ``canopy.<package>.app``, naming the missing extra when the install lacks one.""" 

75 try: 

76 return import_module(f"canopy.{package}.app") 

77 except ModuleNotFoundError as exc: 

78 missing = (exc.name or "").partition(".")[0] 

79 extra = _EXTRA_FOR_PACKAGE.get(missing) 

80 if extra is None: 

81 raise 

82 raise ModuleNotFoundError( 

83 f"The {name} module needs the '{extra}' extra, which is not installed (no module named " 

84 f"'{missing}'). Install canopy[{extra}], or leave {name} out of CANOPY_MODULES; " 

85 "unset, CANOPY_MODULES selects every module.", 

86 name=exc.name, 

87 ) from exc

107def _route_path(scope: Scope) -> str: 

108 """The path the router matches: ``scope["path"]`` without the ``root_path`` prefix. 

109 

110 Behind a path-prefixed ingress (``uvicorn --root-path /canopy``) the server 

111 puts the prefix in front of ``path``; the router strips it, and so must the gate. 

112 """ 

113 path, root_path = scope["path"], scope.get("root_path", "") 

114 if root_path and path.startswith(root_path) and path[len(root_path) : len(root_path) + 1] in ("", "/"): 

115 return path[len(root_path) :] or "/" 

116 return path

119def _is_open(scope: Scope) -> bool: 

120 """``GET /health`` passes the closed-meta gate.""" 

121 return scope["type"] == "http" and scope["method"] == "GET" and _route_path(scope) == "/health"

124class _ClosedMeta: 

125 """Refuse every request without a registry key for the module it targets, except ``GET /health``. 

126 

127 Per-route dependencies cannot do this: the docs and OpenAPI pages of the 

128 parent and of each mounted sub-app, the root document and the per-module 

129 watchdogs have none, and a CORS preflight is answered before routing. 

130 

131 The target module is the one the router would dispatch to: the first route 

132 that fully matches the scope, its sub-app's tag for a mount and ``core`` for 

133 a parent route (or none). A key is admitted iff it is scoped to that module, 

134 whatever its scopes: each route still checks the scope it needs, so a 

135 write-only key can write and an admin-only key can manage keys, as without 

136 the gate. Refusals are the route dependencies' own errors: 401 

137 ``invalid_api_key`` for a missing or unknown key, 503 ``key_store_unavailable`` 

138 when the key store cannot answer and 403 ``insufficient_module_access`` for a 

139 key scoped elsewhere. 

140 

141 The gate sits inside CORSMiddleware and the HTTP metrics middleware. A CORS 

142 preflight, which a browser sends without the key, is answered by CORS and 

143 never reaches the gate; the real request is gated. A refusal carries the CORS 

144 headers, so a browser client can read its 401, and is counted in ``/metrics``. 

145 """ 

146 

147 _DETAIL = "A valid API key is required" 

148 

149 def __init__(self, app: ASGIApp, *, router_app: FastAPI) -> None: 

150 self.app = app 

151 self.router_app = router_app 

152 

153 def _module_for(self, scope: Scope) -> str: 

154 for route in self.router_app.router.routes: 

155 match, _ = route.matches(scope) 

156 if match == Match.FULL: 

157 return module_for_app(route.app) if isinstance(route, Mount) else "core" 

158 return "core" 

159 

160 async def _refusal(self, scope: Scope) -> CanopyError | None: 

161 """The error this scope is refused with, or ``None`` to admit it.""" 

162 token = token_from_scope(scope) 

163 if token is None: 

164 return InvalidAPIKey(self._DETAIL) 

165 try: 

166 key = await get_registry().get(token) 

167 except KeyStoreUnavailable as exc: 

168 return exc 

169 if key is None: 

170 return InvalidAPIKey(self._DETAIL) 

171 module = self._module_for(scope) 

172 if not key.allows_module(module): 

173 return InsufficientModuleAccess(f"API key is not scoped to '{module}'") 

174 return None

176 async def __call__(self, scope: Scope, receive: Receive, send: Send) -> None: 

177 if scope["type"] not in ("http", "websocket"): 

178 await self.app(scope, receive, send) 

179 return 

180 refused = None if _is_open(scope) else await self._refusal(scope) 

181 if refused is None: 

182 await self.app(scope, receive, send) 

183 return 

184 

185 status_code, code, detail = refused.status_code, refused.error_code.value, refused.detail 

186 refusal = JSONResponse({"code": code, "detail": detail}, status_code=status_code) 

187 if scope["type"] == "http": 

188 await refusal(scope, receive, send) 

189 return 

190 websocket = WebSocket(scope, receive, send) 

191 if "websocket.http.response" in scope.get("extensions", {}): 

192 await websocket.send_denial_response(refusal) 

193 else: 

194 # A server that cannot answer a handshake with an HTTP response. 

195 close_code = ( 

196 status.WS_1011_INTERNAL_ERROR 

197 if status_code == status.HTTP_503_SERVICE_UNAVAILABLE 

198 else status.WS_1008_POLICY_VIOLATION 

199 ) 

200 await websocket.close(code=close_code, reason=f"{code}: {detail}")

218def build_app(modules: Iterable[str] | str, *, closed_meta: bool | None = None) -> FastAPI: 

219 """Compose the Canopy API from the named modules. 

220 

221 ``modules`` is an iterable of module names or one comma-separated string; 

222 an empty selection means every module and an unknown name raises 

223 ``ValueError``. Only the selected modules are imported, so a module whose 

224 dependencies are not installed costs nothing as long as it is left out. 

225 

226 ``closed_meta`` puts the whole app behind the key registry except 

227 ``GET /health``; ``None`` takes the ``CANOPY_CLOSED_META`` setting. The key 

228 registry is bound by the lifespan, before anything is served. 

229 ``/v1/api-keys`` and ``/v1/facilities`` are mounted only on an install with 

230 the database key tier (``has_database_tier``). 

231 """ 

232 database_tier = has_database_tier() 

233 if closed_meta is None: 

234 closed_meta = get_settings().closed_meta 

235 

236 selected = parse_modules(modules) if isinstance(modules, str) else resolve_modules(modules) 

237 

238 # Import here, not at module top: an unselected sub-app must never load. A 

239 # module's label is its mount and key scope; its package spells "-" as "_". 

240 packages = {name: name.replace("-", "_") for name in selected} 

241 sub_apps = {name: _import_sub_app(name, package) for name, package in packages.items()} 

242 

243 @asynccontextmanager 

244 async def lifespan(app: FastAPI): 

245 """Compose per-module lifespans. Mounted sub-apps don't run their own.""" 

246 # The static key tier first, then the database when the install has it: 

247 # the long-lived credentials keep working when Postgres is unreachable, 

248 # including on a cold start when the cache has nothing stale to serve. 

249 # Built before anything can serve, so a bad CANOPY_API_KEYS or keys file 

250 # surfaces in the startup log instead of as an unexplained 401. 

251 bind_key_registry(build_key_registry(get_settings())) 

252 if database_tier: 

253 from canopy.core.facilities import gateway as facilities_gateway 

254 

255 facilities_gateway.register_in_use_checker(_facility_granted_to_key, "API key grants") 

256 async with AsyncExitStack() as stack: 

257 for name, module in sub_apps.items(): 

258 await stack.enter_async_context(getattr(module, f"{packages[name]}_lifespan")()) 

259 logger.info(f"Starting Canopy API [version {__version__}] with modules: {', '.join(selected)}") 

260 yield 

261 logger.info("Canopy API shutdown complete") 

262 

263 app = FastAPI( 

264 title="Canopy", 

265 description="Control-system display platform — backend API", 

266 version=__version__, 

267 lifespan=lifespan, 

268 # One aggregate /docs over the selected modules instead (below). 

269 docs_url=None, 

270 redoc_url=None, 

271 ) 

272 app.state.modules = selected 

273 tag_module(app, "core") 

274 

275 if closed_meta: 

276 # Registered first, so it is the innermost layer: CORS answers a preflight 

277 # before it, and its refusals get the CORS headers and are counted. 

278 app.add_middleware(_ClosedMeta, router_app=app) 

279 # Registered before CORSMiddleware so it sits INSIDE it (Starlette wraps in reverse 

280 # registration order): preflight OPTIONS requests that CORS answers itself never 

281 # reach the router, and counting them would only inflate the "unmatched" series. 

282 app.add_middleware(HTTPMetricsMiddleware) 

283 app.add_middleware( 

284 CORSMiddleware, 

285 allow_origins=["*"], 

286 allow_credentials=True, 

287 allow_methods=["*"], 

288 allow_headers=["*"], 

289 ) 

290 app.add_middleware(_SelectedModules, modules=frozenset(selected)) 

291 

292 register_canopy_error_handler(app) 

293 

294 if database_tier: 

295 # Imported here: key and facility management need the database, which a lean install lacks. 

296 from canopy.core.keys.routes import router as api_keys_router 

297 from canopy.core.facilities.routes import router as facilities_router 

298 

299 app.include_router(api_keys_router) 

300 app.include_router(facilities_router) 

301 

302 for name, module in sub_apps.items(): 

303 tag_module(module.app, name) 

304 app.mount(f"/{name}", module.app)
