10"""The Formula Engine's prefix while it was named Fox. Nothing reads it any more."""

13class FormulaEngineConfig(BaseSettings): 

14 """The Formula Engine's configuration. 

15 

16 Sourced from env vars with prefix ``CANOPY_FORMULA_ENGINE__`` and nested delimiter ``__``. 

17 Examples:: 

18 

19 CANOPY_FORMULA_ENGINE__EVAL_TIMEOUT_MS=250 

20 CANOPY_FORMULA_ENGINE__MAX_OUTPUT_POINTS=10000 

21 CANOPY_FORMULA_ENGINE__MAX_EXPRESSION_LEN=2048 

22 """ 

23 

24 model_config = SettingsConfigDict( 

25 env_prefix="CANOPY_FORMULA_ENGINE__", 

26 env_nested_delimiter="__", 

27 env_file=".env", 

28 extra="ignore", 

29 ) 

30 

31 eval_timeout_ms: int = 250 

32 max_output_points: int = 10_000 

33 max_expression_len: int = 2048 

34 

35 # --- resource caps --- # 

36 # Floor on poll-mode interval so a client can't request a tight re-eval loop 

37 # (e.g. intervalMs=1 -> ~1000 evals/sec) that pins a CPU. 

38 min_interval_ms: int = 50 

39 # Max live formula subscriptions a single client may hold at once. 

40 max_subscriptions_per_client: int = 1_000 

41 # Max columns in one ``table`` spec; its bindings may number one more (a 

42 # trigger PV that is not plotted). Above either, the spec is ``invalid_spec``. 

43 table_max_columns: int = 16 

44 # Live table subscriptions one client may hold; the next one gets ``table_limit``. 

45 table_max_per_client: int = 8 

46 # Table row stores kept after their last listener left (for a quick 

47 # re-subscribe); above this the oldest lingering store is dropped. 

48 table_max_lingering: int = 64 

49 # Live table evaluations the whole server runs at once; a subscribe that 

50 # would start one more gets ``table_limit``. Joining a running table (same 

51 # spec and lookback) does not count. Matches ``table_max_lingering``, so at 

52 # most twice this many row stores exist. 

53 table_max_total: int = 64 

54 # Archive samples one input of a table may load while seeding (its trigger 

55 # walk, or its cells). Rows older than where the cap stopped get no value 

56 # for that input, which reports ``truncated``. Matches Vole's per-ring 

57 # ``max_points`` default, the most one PV's history fetch already loads. 

58 table_seed_max_samples: int = Field(default=500_000, ge=1) 

59 

60 # --- live tables --- # 

61 # A table row is final this long (receive clock) after it arrived, or once 

62 # every input has moved past it; newer rows are a provisional tail. 

63 table_settle_horizon_ms: int = 2000 

64 # Poll period of a Trigger PV table, and the floor of a Timer table's Δt. 

65 table_poll_ms: int = 500 

66 # How long an ended table's rows linger for a re-subscribe, the delay before 

67 # a failed history seed is retried, and how long ``lateAfterFinal`` stays set. 

68 table_linger_s: float = 60.0 

69 table_retry_s: float = 60.0 

70 table_late_window_s: float = 60.0 

71 

72 # --- live subscriptions --- # 

73 # A live formula whose inputs have no data yet is retried server-side. Once 

74 # that wait lasts this long, the subscriber gets one error frame flagged 

75 # ``retryable`` (a result later supersedes it). Shorter waits stay silent. 

76 input_wait_notify_s: float = 1.0 

77

84 """Warn once if any ``CANOPY_FOX__*`` variable is still set. 

85 

86 Those variables are ignored since the rename, so a deployment that still 

87 sets them silently runs on the defaults. Called from the module's lifespan, 

88 which runs once per process. 

89 """

95 FormulaEngineConfig.model_config["env_prefix"],
