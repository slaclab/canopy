29@dataclass(frozen=True) 

30class APIKey: 

31 """A resolved key. `name` identifies the holding client class — it is what 

32 request logs and analytics may record; the `token` never is. 

33 

34 `token` is only populated by ``EnvKeyRegistry`` and ``load_keys_file``; ``DbKeyRegistry`` leaves it 

35 empty since nothing downstream reads it back. Excluded from `repr` regardless, 

36 so a stray ``logger.debug("%s", key)`` can never print it. 

37 

38 `modules` has no default: every real registry (``EnvKeyRegistry``, 

39 ``DbKeyRegistry``) already passes it explicitly, so a future one that 

40 forgets to must fail loudly rather than silently granting every module. 

41 """ 

42 

43 read: bool 

44 write: bool 

45 modules: frozenset[str] 

46 admin: bool = False 

47 name: str = "unknown" 

48 id: str | None = None 

49 token: str = field(default="", repr=False) 

50 facilities: frozenset[str] = frozenset() 

51 

52 def allows_module(self, module: str) -> bool: 

53 return module == JOBS or WILDCARD in self.modules or module in self.modules 

54 

55 @property 

56 def unrestricted_facilities(self) -> bool: 

57 """Whether this key may reach every facility's data. 

58 

59 There is no ``allows_facility(id)`` counterpart to ``allows_module``: 

60 ``facilities`` holds slugs, rows carry ids, and a grant on a parent covers its 

61 descendants, so the check needs hierarchy expansion against the database. 

62 That resolution lives in ``core.facilities.gateway``; the per-row check lives 

63 in the module owning the row. 

64 """ 

65 return WILDCARD in self.facilities

91class EnvKeyRegistry(_DictRegistry): 

92 """In-memory registry built from a comma-separated env-var entry list. 

93 

94 Format: ``"<token>:<scope>[+<scope>]*[@<module>[+<module>]*][,...]"``. 

95 Scopes are ``read``, ``write`` and ``admin``. The ``@<module>...`` suffix 

96 restricts the key to those modules; pass ``@*`` for every module. An entry 

97 with no ``@`` at all is deprecated but not fatal — it defaults to ``@*`` 

98 with a warning, so an env var written before module scoping existed keeps 

99 granting the same access it always did. Examples: 

100 

101 CANOPY_API_KEYS="" # empty registry 

102 CANOPY_API_KEYS="dev:read+write@*" # one key, full access 

103 CANOPY_API_KEYS="ro:read@*,rw:read+write@*" # two keys 

104 CANOPY_API_KEYS="ops:admin@*" # key management only 

105 CANOPY_API_KEYS="ops:admin@core" # admin, core routes only 

106 CANOPY_API_KEYS="ui:read+write@squirrel+formula-engine" # scoped to two modules 

107 

108 There is no facility syntax here, by decision rather than omission: every env key 

109 grants every facility. Resolving a grant requires reading the hierarchy from the 

110 database, which is exactly what this tier has to survive without. Restrict by 

111 facility with a database-backed key. 

112 

113 Env keys are no longer a stop-gap: they are the statically stable layer in 

114 front of the Postgres-backed registry. Keys defined here keep working when the 

115 database is unreachable — including on a cold start, when there is nothing 

116 cached to fall back on — so a deployment keeps a small, long-lived set here 

117 (a read key for displays, an admin key for ops) and manages everything else in 

118 the database, where revocation is immediate. 

119 

120 A typo'd entry is skipped rather than fatal, so parsing logs a warning for 

121 each one it drops — an unreported skip means a deployed instance rejects a 

122 client for a reason nothing on the box explains. Entries are identified by 

123 position, never by content: the token is the secret. 

124 """ 

125 

126 def __init__(self, raw: str = "") -> None: 

127 self._keys: dict[str, APIKey] = {} 

128 for position, entry in enumerate((e.strip() for e in raw.split(",")), start=1): 

129 if not entry: 

130 continue 

131 try: 

132 token, rest = entry.split(":", 1) 

133 except ValueError: 

134 logger.warning( 

135 "Skipping malformed CANOPY_API_KEYS entry #%d: expected '<token>:<scope>[+<scope>]'", position 

136 ) 

137 continue 

138 token = token.strip() 

139 if not token: 

140 logger.warning("Skipping CANOPY_API_KEYS entry #%d: empty token", position) 

141 continue 

142 scopes, has_at, modules_part = rest.partition("@") 

143 scope_set = {s.strip() for s in scopes.split("+")} 

144 if unknown := scope_set - {"read", "write", "admin"}: 

145 logger.warning( 

146 "Skipping CANOPY_API_KEYS entry #%d: %d unrecognized scope(s); " 

147 "only 'read', 'write' and 'admin' are valid", 

148 position, 

149 len(unknown), 

150 ) 

151 continue 

152 if not has_at: 

153 logger.warning( 

154 "CANOPY_API_KEYS entry #%d has no '@<module>' suffix; defaulting to '@*' (every module). " 

155 "This default is deprecated — add '@*' explicitly, or scope the key, before it is removed.", 

156 position, 

157 ) 

158 modules_part = WILDCARD 

159 module_set = {m.strip() for m in modules_part.split("+") if m.strip()} 

160 if not module_set: 

161 logger.warning("Skipping CANOPY_API_KEYS entry #%d: empty '@' module list", position) 

162 continue 

163 known_modules, unknown_modules = split_modules(module_set) 

164 if unknown_modules: 

165 logger.warning( 

166 "CANOPY_API_KEYS entry #%d has %d unrecognized module(s); ignoring them", 

167 position, 

168 len(unknown_modules), 

169 ) 

170 if not known_modules: 

171 logger.warning( 

172 "Skipping CANOPY_API_KEYS entry #%d: no recognized modules remained after filtering", position 

173 ) 

174 continue 

175 if token in self._keys: 

176 logger.warning("CANOPY_API_KEYS entry #%d duplicates an earlier token; the later entry wins", position) 

177 self._keys[token] = APIKey( 

178 token=token, 

179 read="read" in scope_set, 

180 write="write" in scope_set, 

181 admin="admin" in scope_set, 

182 # Env keys have no app name of their own; position identifies them in 

183 # logs without ever revealing the token. 

184 name=f"env:{position}", 

185 modules=known_modules, 

186 # Every facility, unconditionally — see the class docstring. 

187 facilities=frozenset({WILDCARD}), 

188 ) 

189 if self._keys: 

190 logger.info( 

191 "Registered %d API key(s) from CANOPY_API_KEYS (%d read, %d write, %d admin, %d module-scoped)", 

192 len(self._keys), 

193 sum(1 for key in self._keys.values() if key.read), 

194 sum(1 for key in self._keys.values() if key.write), 

195 sum(1 for key in self._keys.values() if key.admin), 

196 sum(1 for key in self._keys.values() if WILDCARD not in key.modules), 

197 ) 

198 else: 

199 logger.info("No API keys registered from CANOPY_API_KEYS; relying on the database registry")

220class _BadFileEntry(ValueError): 

221 """A keys-file entry breaks the grammar; the message names the rule broken.""" 

222 

223 

224def _parse_file_entry(entry: str) -> tuple[str, frozenset[str], frozenset[str]]: 

225 """``entry`` as ``(token, scopes, recognized modules)``, or `_BadFileEntry` saying why not. 

226 

227 The reason names the rule broken, never a part of the entry: the token is 

228 the secret, and a scope or module typo sits right next to it. 

229 """ 

230 if entry.startswith("#"): 

231 # A commented-out key would otherwise become a live key whose token starts with "#". 

232 raise _BadFileEntry("starts with # (a keys file has no comments)") 

233 if any(char.isspace() for char in entry): 

234 raise _BadFileEntry("contains whitespace") 

235 token, separator, rest = entry.partition(":") 

236 if not separator or not token: 

237 raise _BadFileEntry("is not <token>:<scope>[+<scope>]@<module>[+<module>]") 

238 scopes, at, modules = rest.partition("@") 

239 if not at: 

240 raise _BadFileEntry("has no @<module>[+<module>] suffix") 

241 scope_list = scopes.split("+") 

242 if not all(scope_list) or not set(scope_list) <= KEYS_FILE_SCOPES: 

243 raise _BadFileEntry("names a scope other than read, write or admin") 

244 if "read" not in scope_list: 

245 raise _BadFileEntry("grants no read scope") 

246 module_list = modules.split("+") 

247 if not all(module_list): 

248 raise _BadFileEntry("has an empty module in its @<module>[+<module>] suffix") 

249 known, unknown = split_modules(set(module_list)) 

250 if unknown: 

251 raise _BadFileEntry("names an unknown module") 

252 return token, frozenset(scope_list), known

255def load_keys_file(path: str | Path) -> FileKeyRegistry: 

256 """Build the registry from a keys file. 

257 

258 Each entry is ``<token>:<scope>[+<scope>]@<module>[+<module>]``: scopes 

259 ``read``, ``write`` and ``admin``, ``read`` always among them, and modules 

260 from ``canopy.core.modules`` (``*`` for every module). Entries are separated 

261 by commas or line breaks. Unlike ``CANOPY_API_KEYS`` the file is read 

262 strictly: a file that cannot be read, holds no key, or holds one entry that 

263 breaks the grammar raises ``KeysFileError`` naming the entry's position, and 

264 there is no default module scope. The file has no comment syntax: an entry 

265 starting with ``#`` is refused rather than read as a token. A deployment that names a 

266 file never falls back to ``CANOPY_API_KEYS``; an empty ``CANOPY_API_KEYS_FILE`` 

267 names no file and counts as unset. 

268 

269 Like an env key, a file key grants every facility: the static tier has no 

270 facility syntax (see ``EnvKeyRegistry``). Imports nothing from 

271 ``canopy.core.keys``, so a lean install can read a keys file. 

272 """ 

273 try: 

274 # utf-8-sig: an editor that saves a BOM must not glue it to the first token. 

275 raw = Path(path).read_text(encoding="utf-8-sig") 

276 except (OSError, ValueError) as exc: 

277 reason = exc.strerror if isinstance(exc, OSError) and exc.strerror else type(exc).__name__ 

278 raise KeysFileError(f"CANOPY_API_KEYS_FILE {path}: cannot be read ({reason})") from None 

279 entries = [entry.strip() for entry in raw.replace("\r", "\n").replace("\n", ",").split(",") if entry.strip()] 

280 if not entries: 

281 raise KeysFileError(f"CANOPY_API_KEYS_FILE {path}: holds no API key") 

282 keys: dict[str, APIKey] = {} 

283 for number, entry in enumerate(entries, start=1): 

284 try: 

285 token, scopes, modules = _parse_file_entry(entry) 

286 except _BadFileEntry as problem: 

287 raise KeysFileError(f"CANOPY_API_KEYS_FILE {path}: entry {number} {problem}") from None 

288 if token in keys: 

289 raise KeysFileError(f"CANOPY_API_KEYS_FILE {path}: entry {number} repeats an earlier token") 

290 keys[token] = APIKey( 

291 read=True, 

292 write="write" in scopes, 

293 admin="admin" in scopes, 

294 modules=modules, 

295 # Position identifies a file key in logs without revealing its token. 

296 name=f"file:{number}", 

297 token=token, 

298 # Every facility, as for an env key: the static tier has no facility syntax. 

299 facilities=frozenset({WILDCARD}), 

300 ) 

301 logger.info( 

302 "Registered %d API key(s) from CANOPY_API_KEYS_FILE (%d write, %d admin, %d module-scoped)", 

303 len(keys), 

304 sum(1 for key in keys.values() if key.write), 

305 sum(1 for key in keys.values() if key.admin), 

306 sum(1 for key in keys.values() if WILDCARD not in key.modules), 

307 ) 

308 return FileKeyRegistry(keys)

457def require_wildcard_facility_access(key: AdminKeyDep) -> APIKey: 

458 """Gate for editing the facility hierarchy itself, not just a key's grant.

484async def ws_get_api_key(websocket: WebSocket) -> APIKey: 

485 token = websocket.headers.get("X-API-Key") or websocket.query_params.get("api_key") 

486 if not token:

493 except KeyStoreUnavailable:

503 reason=f"{ErrorCode.INVALID_API_KEY.value}: invalid API key",

506 if not key.allows_module(module): 

507 raise WebSocketException( 

508 code=status.WS_1008_POLICY_VIOLATION, 

509 reason=f"{ErrorCode.INSUFFICIENT_MODULE_ACCESS.value}: API key is not scoped to '{module}'", 

510 )

514def token_from_scope(scope: Scope) -> str | None: 

515 """The API key token carried by a raw ASGI scope, or ``None``. 

516 

517 For code that runs outside dependency injection. Reads what the route 

518 dependencies read: the ``X-API-Key`` header, and for a WebSocket also the 

519 ``api_key`` query parameter (browsers cannot set headers on a handshake). 

520 """ 

521 token = next( 

522 (value.decode("latin-1") for name, value in scope.get("headers", ()) if name.lower() == b"x-api-key"), 

523 None, 

524 ) 

525 if not token and scope["type"] == "websocket": 

526 query = parse_qs(scope.get("query_string", b"").decode("latin-1")) 

527 token = next(iter(query.get("api_key", ())), None) 

528 return token or None
