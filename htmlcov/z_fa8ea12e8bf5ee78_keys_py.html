1"""Redis keys and the entry format shared by the PV cache's writer and readers. 

2 

3The elected ``canopy-worker`` writes; snapshot create and the health endpoint read. 

4Each cached value is stored as ``{"gen": <lease generation>, "cachedAt": <unix seconds>, 

5"pv": <PVValue>}`` so a reader gets back exactly the ``PVValue`` a live 

6``connector.get()`` would return, and can tell which leader's lease wrote it. The 

7leader's status report is a `PVCacheReport`, and `untrusted_reason` is the one check 

8of it that snapshot create and the health endpoint share. 

9""" 

10 

11from __future__ import annotations 

12 

13import json 

14from typing import Any 

15 

16from pydantic import ConfigDict, ValidationError 

17 

18from canopy.contracts.pv import PVValue 

19from canopy.contracts.base import CanopyModel 

20 

21# Hash: PV address (as registered, protocol prefix included) -> JSON entry. 

22VALUES_KEY = "canopy:squirrel:pv_cache:values" 

23# String: the leader's instance id, with a lease TTL. 

24LEADER_KEY = "canopy:squirrel:pv_cache:leader" 

25# String: the leader's JSON report, rewritten on every flush (its heartbeat). 

26STATUS_KEY = "canopy:squirrel:pv_cache:status" 

27# How long a report outlives its last write: long enough for /health/pv-cache to show 

28# who led last and when, and short of keeping it forever once the cache is turned off. 

29STATUS_TTL_MS = 60 * 60 * 1000 

30 

31# Entries per HSET / HMGET call, so a full 40k-PV warm-up isn't one multi-MB command. 

32BATCH_SIZE = 2000 

33 

34 

35def encode_entry(value: PVValue, cached_at: float, generation: str | None) -> str: 

36 return json.dumps({"gen": generation, "cachedAt": cached_at, "pv": value.model_dump(mode="json")}) 

37 

38 

39def decode_entry(raw: bytes | str) -> tuple[str | None, PVValue]: 

40 """The generation that wrote the entry, and its value.""" 

41 entry = json.loads(raw) 

42 return entry.get("gen"), PVValue.model_validate(entry["pv"])

45class PVCacheReport(CanopyModel): 

46 """The leader's status report, rewritten on every flush (camelCase on the wire). 

47 

48 ``heartbeat_at`` is the flush time, or 0 once the leader has stepped down. The 

49 counts are described on `PVCacheStatusDTO`. 

50 """ 

51 

52 model_config = ConfigDict(allow_inf_nan=False) 

53 

54 instance: str | None = None 

55 host: str | None = None 

56 generation: str | None = None 

57 leader_since: float | None = None 

58 heartbeat_at: float 

59 configured: int | None = None 

60 monitored: int | None = None 

61 connected: int | None = None 

62 seen: int | None = None 

63 warm: bool | None = None

66class UnreadableStatus(ValueError): 

67 """``STATUS_KEY`` holds something that isn't a leader's report.""" 

68 

69 

70async def read_status(redis: Any) -> PVCacheReport | None: 

71 """The leader's last report, or ``None`` when there is none. 

72 

73 Raises `UnreadableStatus` when the key holds anything else, so a caller can fall 

74 back as it would with no report instead of failing on it. 

75 """ 

76 raw = await redis.get(STATUS_KEY) 

77 if raw is None: 

78 return None 

79 try: 

80 return PVCacheReport.model_validate_json(raw) 

81 except ValidationError as exc: 

82 raise UnreadableStatus(f"the PV cache report is unreadable ({exc.error_count()} invalid field(s))") from exc 

83

85def untrusted_reason(report: PVCacheReport, now: float, max_age_s: float) -> str | None: 

86 """Why the cache this report describes mustn't be read, or ``None`` when it may. 

87 

88 Snapshot create skips the whole cache for any of these; the health endpoint 

89 reports the same reason as ``down``. 

90 """ 

91 if report.heartbeat_at == 0: 

92 return "the last PV cache leader stepped down and no worker has taken over" 

93 age = now - report.heartbeat_at 

94 if age > max_age_s: 

95 return f"the PV cache heartbeat is {age:.0f}s old (limit {max_age_s:g}s)" 

96 if not report.generation: 

97 return "the PV cache report names no leader generation" 

98 return None
