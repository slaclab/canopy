1"""Simulation connector backend for testing and easy development. 

2 

3Default backend so the whole live-PV path runs and tests anywhere, with no EPICS 

4infrastructure. The PV name's suffix selects the value kind so every type can be 

5exercised end to end: 

6 

7- ``:ENUM`` -> an enum cycling through fixed choices 

8- ``:WAVEFORM`` / ``:WF`` -> a numeric waveform (animated sine across the array) 

9- ``:STR`` -> a string value 

10- ``:TABLE`` -> a small NTTable-style table (index/value/status columns) 

11- anything else -> a scalar 

12 

13Each PV's phase is a stable function of its name, so values are deterministic per 

14process. Severity thresholds let the alarm-styling path be exercised too, and every 

15value carries the EPICS alarm status an IOC would report with it: HIHI / HIGH above 

16the thresholds, LOLO / LOW below them, and STATE for the enum's fault choice. 

17 

18Writes can be made to fail on demand — for a named set of PVs, or at a configured 

19rate drawn from a seeded RNG — so a caller's partial-failure handling can be 

20exercised in the dev stack without an IOC. 

21"""

63class SimulationConnector: 

64 """In-process `PVConnector` that generates values instead of talking to an IOC. 

65 

66 Values written through `put` are held in memory and returned by subsequent reads, 

67 standing in for an IOC's record state. Writes can also be made to fail, so a 

68 caller's partial-failure handling is reachable without real hardware. 

69 """ 

70 

71 backend_name = "simulation" 

72 

73 def __init__( 

74 self, 

75 update_hz: float = 5.0, 

76 units: str | None = None, 

77 waveform_length: int = 64, 

78 max_array_elements: int = 100_000, 

79 put_error_rate: float = 0.0, 

80 put_fail_pvs: frozenset[str] = frozenset(), 

81 seed: int | None = None, 

82 ) -> None: 

83 """Build a simulation connector. 

84 

85 Parameters 

86 ---------- 

87 update_hz : float, optional 

88 Monitor tick rate, in updates per second. 

89 units : str | None, optional 

90 Engineering units reported by `describe` for scalar and waveform PVs. 

91 waveform_length : int, optional 

92 Element count of simulated waveform PVs, capped by `max_array_elements`. 

93 max_array_elements : int, optional 

94 Ceiling on the length of a generated array. 

95 put_error_rate : float, optional 

96 Fraction of writes to fail, in ``[0.0, 1.0]``. 

97 put_fail_pvs : frozenset[str], optional 

98 PV names whose writes always fail, whatever `put_error_rate` is. Matched 

99 without their ``ca://`` / ``pva://`` prefix, on both sides. 

100 seed : int | None, optional 

101 Seed for this instance's private failure RNG. `None` seeds from the 

102 system entropy source. 

103 """ 

104 self._interval = 1.0 / (update_hz or 5.0) 

105 self._units = units or None 

106 self._waveform_length = max(1, min(waveform_length, max_array_elements)) 

107 self._tasks: set[asyncio.Task] = set() 

108 self._overrides: dict[str, PVWriteValue] = {} 

109 self._put_error_rate = put_error_rate 

110 self._put_fail_pvs = frozenset(strip_protocol_prefix(name) for name in put_fail_pvs) 

111 self._rng = random.Random(seed)

153 @staticmethod 

154 def _kind_for(pv: str) -> ValueKind: 

155 if pv.endswith(":ENUM"): 

156 return ValueKind.ENUM 

157 if pv.endswith((":WAVEFORM", ":WF")): 

158 return ValueKind.WAVEFORM 

159 if pv.endswith(":STR"): 

160 return ValueKind.STRING 

161 if pv.endswith(":TABLE"): 

162 return ValueKind.TABLE 

163 return ValueKind.SCALAR 

164 

165 @staticmethod 

166 def _phase(pv: str) -> float: 

167 return (zlib.crc32(pv.encode()) % 1000) / 1000.0 * 2 * math.pi 

168 

169 def _sine(self, pv: str) -> float: 

170 return round(10.0 * math.sin(2 * math.pi * 0.1 * time.monotonic() + self._phase(pv)), 4) 

171 

172 @staticmethod 

173 def _severity_for(mag: float) -> AlarmSeverity: 

174 if mag > 9.0: 

175 return AlarmSeverity.MAJOR 

176 if mag > 7.0: 

177 return AlarmSeverity.MINOR 

178 return AlarmSeverity.NO_ALARM

180 @staticmethod 

181 def _alarm_for(v: float) -> tuple[AlarmSeverity, int]: 

182 """The severity and alarm status an IOC would report with a simulated scalar. 

183 

184 Parameters 

185 ---------- 

186 v : float 

187 The generated value. 

188 

189 Returns 

190 ------- 

191 tuple[AlarmSeverity, int] 

192 MAJOR with HIHI or LOLO past the outer threshold, MINOR with HIGH or LOW 

193 past the inner one, NO_ALARM otherwise. The status direction follows the 

194 sign of ``v``. 

195 """ 

196 severity = SimulationConnector._severity_for(abs(v)) 

197 if severity is AlarmSeverity.MAJOR: 

198 return severity, ALARM_STATUS_CODES["HIHI" if v > 0 else "LOLO"] 

199 if severity is AlarmSeverity.MINOR: 

200 return severity, ALARM_STATUS_CODES["HIGH" if v > 0 else "LOW"] 

201 return severity, _NO_ALARM_STATUS 

202 

203 @staticmethod 

204 def _enum_alarm(choice: object) -> tuple[AlarmSeverity, int]: 

205 """The severity and alarm status an IOC would report with a simulated enum choice. 

206 

207 Parameters 

208 ---------- 

209 choice : object 

210 The choice, as its index into the enum's choices or as its name. 

211 

212 Returns 

213 ------- 

214 tuple[AlarmSeverity, int] 

215 MAJOR with STATE for the FAULT choice, NO_ALARM otherwise. 

216 """ 

217 if choice in (_FAULT_INDEX, "FAULT"): 

218 return AlarmSeverity.MAJOR, ALARM_STATUS_CODES["STATE"] 

219 return AlarmSeverity.NO_ALARM, _NO_ALARM_STATUS

221 def _value_for(self, pv: str) -> PVValue: 

222 """Generate the current value of a simulated PV. 

223 

224 A PV pinned by a prior `put` is returned verbatim, keeping the name-derived 

225 value kind so a write round-trips through a read, with the alarm an IOC would 

226 raise for that value. A ``ca://`` / ``pva://`` prefix doesn't change which PV 

227 is read: every spelling shares one generated value and one written override. 

228 

229 Parameters 

230 ---------- 

231 pv : str 

232 PV name; its suffix selects the value kind. 

233 

234 Returns 

235 ------- 

236 PVValue 

237 The value, its kind, and an alarm severity with the matching status. 

238 """ 

239 name = strip_protocol_prefix(pv) 

240 kind = self._kind_for(name) 

241 severity, status = AlarmSeverity.NO_ALARM, _NO_ALARM_STATUS 

242 if name in self._overrides: 

243 value = self._overrides[name] 

244 if kind is ValueKind.SCALAR and isinstance(value, int | float): 

245 severity, status = self._alarm_for(float(value)) 

246 elif kind is ValueKind.ENUM: 

247 severity, status = self._enum_alarm(value) 

248 elif kind is ValueKind.ENUM: 

249 value = int(time.monotonic() / 2) % len(_ENUM_CHOICES) 

250 severity, status = self._enum_alarm(value) 

251 elif kind is ValueKind.WAVEFORM: 

252 t = 2 * math.pi * 0.1 * time.monotonic() + self._phase(name) 

253 n = self._waveform_length 

254 value = [round(10.0 * math.sin(t + i * 2 * math.pi / n), 4) for i in range(n)] 

255 elif kind is ValueKind.STRING: 

256 value = f"sample={self._sine(name):+.3f}" 

257 elif kind is ValueKind.TABLE: 

258 n = 4 

259 base = self._sine(name) 

260 rows = list(range(1, n + 1)) 

261 vals = [round(base + i, 3) for i in range(n)] 

262 states = [_ENUM_CHOICES[(i + int(time.monotonic())) % len(_ENUM_CHOICES)] for i in range(n)] 

263 value = TableValue(labels=["Index", "Value", "Status"], columns=[rows, vals, states]) 

264 else: 

265 value = self._sine(name) 

266 severity, status = self._alarm_for(value) 

267 return PVValue( 

268 pv=pv, 

269 value=value, 

270 value_kind=kind, 

271 severity=severity, 

272 alarm_status=status, 

273 ts_ns=time.time_ns(), 

274 )

279 async def put(self, pv: str, value: PVWriteValue, *, timeout: float | None = None, wait: bool = True) -> None: 

280 """Pin a PV's value so subsequent reads and monitor ticks return it. 

281 

282 The override is held under the name without any ``ca://`` / ``pva://`` prefix, 

283 so a read through any spelling of the PV returns it. It is never evicted, which is bounded in practice by the small set 

284 of distinct PVs a dev or test session writes to. 

285 

286 Parameters 

287 ---------- 

288 pv : str 

289 PV name to write. 

290 value : PVWriteValue 

291 Value to hold as this PV's current value. 

292 timeout : float | None, optional 

293 Accepted for protocol compatibility; the write completes immediately. 

294 wait : bool, optional 

295 Accepted for protocol compatibility; the write completes immediately. 

296 

297 Raises 

298 ------ 

299 PVWriteFailed 

300 When `pv`, ignoring any ``ca://`` / ``pva://`` prefix, is one of the 

301 configured always-failing PVs, or when the instance's RNG draws below the 

302 configured put error rate. 

303 """ 

304 name = strip_protocol_prefix(pv) 

305 # Draw on every put, so naming a PV to always fail doesn't shift the seeded 

306 # outcomes of the others. 

307 drawn = self._rng.random() 

308 if name in self._put_fail_pvs or drawn < self._put_error_rate: 

309 raise PVWriteFailed(f"simulated put failure for {pv}") 

310 self._overrides[name] = value

312 async def describe(self, pv: str) -> PVMeta: 

313 """Describe a simulated PV's static metadata. 

314 

315 The metadata follows from the value kind its name suffix selects. A ``ca://`` / 

316 ``pva://`` prefix doesn't change it: every spelling of a PV is described alike. 

317 

318 Parameters 

319 ---------- 

320 pv : str 

321 PV name; its suffix selects the value kind. 

322 

323 Returns 

324 ------- 

325 PVMeta 

326 The kind's units, limits, precision and enum choices, with `pv` echoing 

327 the name as requested. 

328 """ 

329 name = strip_protocol_prefix(pv) 

330 kind = self._kind_for(name) 

331 if kind is ValueKind.ENUM: 

332 return PVMeta(pv=pv, value_kind=kind, enum_choices=list(_ENUM_CHOICES), writable=True) 

333 if kind is ValueKind.WAVEFORM: 

334 return PVMeta(pv=pv, value_kind=kind, units=self._units, display_low=-10.0, display_high=10.0) 

335 if kind is ValueKind.STRING: 

336 return PVMeta(pv=pv, value_kind=kind, writable=True) 

337 if kind is ValueKind.TABLE: 

338 return PVMeta(pv=pv, value_kind=kind) 

339 return PVMeta( 

340 pv=pv, 

341 value_kind=kind, 

342 units=self._units, 

343 precision=4, 

344 display_low=-10.0, 

345 display_high=10.0, 

346 control_low=-10.0, 

347 control_high=10.0, 

348 warning_low=-7.0, 

349 warning_high=7.0, 

350 alarm_low=-9.0, 

351 alarm_high=9.0, 

352 writable=True, 

353 )

357 async def subscribe(self, pv: str, on_event: EventCallback) -> Subscription: 

358 task = asyncio.create_task(self._run(pv, on_event)) 

359 self._tasks.add(task) 

360 task.add_done_callback(self._tasks.discard) 

361 return _SimSubscription(task) 

362 

363 async def _run(self, pv: str, on_event: EventCallback) -> None: 

364 # Emit an initial value immediately, then once per tick. 

365 on_event(self._value_for(pv)) 

366 while True: 

367 await asyncio.sleep(self._interval) 

368 on_event(self._value_for(pv))
