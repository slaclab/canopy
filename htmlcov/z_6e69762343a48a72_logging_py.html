1"""Application logging for both process types (canopy-api and canopy-worker). 

2 

3Configured from core settings (``CANOPY_LOG_FORMAT`` / ``CANOPY_LOG_PATH`` / 

4``CANOPY_LOG_LEVEL``). The "json" format emits one JSON object per line with 

5any ``extra={...}`` attributes merged in as first-class fields, so a log 

6shipper (Promtail in ``docker/observability/``) can parse lines with a plain 

7``json`` pipeline stage. File output is best-effort: if the log file or its 

8directory cannot be created, logging falls back to stdout only rather than 

9failing startup. 

10 

11Framework loggers (uvicorn, arq) are folded into the same root handlers, so 

12HTTP access lines and worker lifecycle messages land in the same file, in the 

13same format, as application logs. 

14""" 

15 

16import sys 

17import json 

18import logging 

19import traceback 

20from pathlib import Path 

21from datetime import UTC, datetime 

22from logging.handlers import RotatingFileHandler 

23 

24from canopy.core.config import get_settings 

25 

26# Rotation bounds for the file log. It exists only for a log shipper to tail, 

27# so a short window is enough - and without rotation a log path that isn't on 

28# a mounted volume grows in the container's writable layer for the life of the 

29# container. 

30_LOG_MAX_BYTES = 50 * 1024 * 1024 

31_LOG_BACKUP_COUNT = 3 

32 

33# Attributes every LogRecord carries; anything else on the record came from 

34# `extra={...}` at the call site and is merged into the JSON line. 

35_STANDARD_RECORD_ATTRS = frozenset(vars(logging.LogRecord("", 0, "", 0, "", (), None)).keys()) | { 

36 "message", 

37 "asctime", 

38 "taskName", 

39 # uvicorn attaches an ANSI-coloured duplicate of `message` for its own 

40 # terminal formatter; it carries nothing the JSON line needs. 

41 "color_message", 

42} 

43 

44# Keys JsonFormatter writes itself. An `extra` that collides is written under 

45# an `extra_` prefix rather than allowed to overwrite: `level` is a Promtail 

46# index label, so a collision would corrupt querying, not just the one line. 

47_RESERVED_KEYS = frozenset({"ts", "level", "logger", "message", "error"}) 

48 

49# Loggers that uvicorn / arq configure with their own handlers (uvicorn's with 

50# propagate=False) before the app is imported; see _adopt_framework_loggers. 

51_FRAMEWORK_LOGGERS = ("uvicorn", "uvicorn.error", "uvicorn.access", "arq") 

52 

53_TEXT_FORMAT = "%(asctime)s [%(levelname)s] %(name)s: %(message)s" 

54 

55 

56class JsonFormatter(logging.Formatter): 

57 """One JSON object per line: ts / level / logger / message + extras.""" 

58 

59 def format(self, record: logging.LogRecord) -> str: 

60 payload: dict = { 

61 "ts": datetime.fromtimestamp(record.created, tz=UTC).isoformat(), 

62 "level": record.levelname, 

63 "logger": record.name, 

64 "message": record.getMessage(), 

65 } 

66 for key, value in record.__dict__.items(): 

67 if key in _STANDARD_RECORD_ATTRS or key.startswith("_"): 

68 continue 

69 payload[f"extra_{key}" if key in _RESERVED_KEYS else key] = value 

70 if record.exc_info and record.exc_info[0] is not None: 

71 payload["error"] = "".join(traceback.format_exception(*record.exc_info)).rstrip() 

72 return json.dumps(payload, default=str) 

73

75def _resolve_level(name: str) -> int | None: 

76 """Map a log level name to its number. 

77 

78 Parameters 

79 ---------- 

80 name : str 

81 Level name, matched case-insensitively and ignoring surrounding whitespace. 

82 

83 Returns 

84 ------- 

85 int | None 

86 The level number, or ``None`` when ``name`` is not a known level. 

87 """ 

88 return logging.getLevelNamesMapping().get(name.strip().upper())

91def configure_logging(level: int | None = None) -> None: 

92 """Configure application logging. Idempotent, safe to call repeatedly. 

93 

94 Parameters 

95 ---------- 

96 level : int | None 

97 Level number that overrides ``CANOPY_LOG_LEVEL`` when given, kept for callers 

98 that predate the settings-driven config. When omitted, ``CANOPY_LOG_LEVEL`` is 

99 used; an unknown name there falls back to ``INFO`` and logs a warning through 

100 the configured handlers instead of failing process startup. 

101 

102 Returns 

103 ------- 

104 None 

105 """ 

106 settings = get_settings() 

107 unknown_level: str | None = None 

108 if level is None: 

109 resolved = _resolve_level(settings.log_level) 

110 if resolved is None: 

111 unknown_level = settings.log_level 

112 level = logging.INFO if resolved is None else resolved 

113 

114 formatter: logging.Formatter 

115 if settings.log_format == "json": 

116 formatter = JsonFormatter() 

117 else: 

118 formatter = logging.Formatter(_TEXT_FORMAT) 

119 

120 handlers: list[logging.Handler] = [logging.StreamHandler(sys.stdout)] 

121 file_error: OSError | None = None 

122 if settings.log_path: 

123 try: 

124 log_path = Path(settings.log_path) 

125 log_path.parent.mkdir(parents=True, exist_ok=True) 

126 handlers.append( 

127 RotatingFileHandler(log_path, maxBytes=_LOG_MAX_BYTES, backupCount=_LOG_BACKUP_COUNT, encoding="utf-8") 

128 ) 

129 except OSError as exc: 

130 file_error = exc 

131 

132 for handler in handlers: 

133 handler.setFormatter(formatter) 

134 logging.basicConfig(level=level, handlers=handlers, force=True) 

135 _adopt_framework_loggers() 

136 

137 if file_error is not None: 

138 # Best-effort: a bad log path must not fail process startup. Reported 

139 # after basicConfig so the warning goes through the configured handlers 

140 # (structured, in JSON mode) instead of logging.lastResort on stderr. 

141 logging.getLogger(__name__).warning( 

142 "Could not open log file %s (%s); logging to stdout only", settings.log_path, file_error 

143 ) 

144 if unknown_level is not None: 

145 logging.getLogger(__name__).warning("Unknown log level %r, falling back to INFO", unknown_level) 

146

148def _adopt_framework_loggers() -> None: 

149 """Route uvicorn's and arq's loggers through the root handlers. 

150 

151 Both frameworks apply their own dictConfig before importing the app: 

152 uvicorn attaches stderr/stdout handlers to ``uvicorn`` and 

153 ``uvicorn.access`` with ``propagate=False`` (so access logs would never 

154 reach the JSON file), and arq attaches a stdout handler to ``arq`` (so its 

155 lines would print twice). Dropping those handlers and letting the records 

156 propagate puts everything through one formatter, one file, and one level 

157 (``CANOPY_LOG_LEVEL``). 

158 """ 

159 for name in _FRAMEWORK_LOGGERS: 

160 framework_logger = logging.getLogger(name) 

161 for handler in framework_logger.handlers: 

162 handler.close() 

163 framework_logger.handlers.clear() 

164 framework_logger.propagate = True 

165 framework_logger.setLevel(logging.NOTSET)
