24# Unguessable public token. base62 avoids URL-encoding surprises; length 10 gives 

25# 62^10 ≈ 8.4e17 values, so the collision-retry loop below is just a safety net. 

26_SLUG_ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ" 

27_SLUG_LENGTH = 10 

28_SLUG_MAX_RETRIES = 5

31def _new_slug() -> str: 

32 return "".join(secrets.choice(_SLUG_ALPHABET) for _ in range(_SLUG_LENGTH))

35def _check_text(value: str | None, field: str) -> None: 

36 """Reject NUL in a text column, which Postgres cannot store (see _check_data).""" 

37 if value is not None and "\x00" in value: 

38 raise BadRequest(f"Config {field} must not contain NUL characters")

41def _has_nul(value: object) -> bool: 

42 """Recursively test a decoded JSON value for a NUL in any string (key or value).""" 

43 if isinstance(value, str): 

44 return "\x00" in value 

45 if isinstance(value, dict): 

46 return any(_has_nul(k) or _has_nul(v) for k, v in value.items()) 

47 if isinstance(value, list | tuple): 

48 return any(_has_nul(v) for v in value) 

49 return False

52class ConfigService: 

53 """Config store CRUD. Owns no transaction — the caller commits. 

54 

55 Every read/mutate is scoped by ``app``: the slug locates the row (slugs are 

56 globally unique), then the service asserts the row's ``app`` matches the 

57 expected one, treating a mismatch as "not found". This app handshake keeps a 

58 consumer from accidentally loading another app's config. 

59 """ 

60 

61 def __init__(self, session: AsyncSession): 

62 self.session = session 

63 self.repo = ConfigRepository(session)

71 description=config.description, 

72 data=config.data, 

73 created_by=config.created_by, 

74 created_date=config.created_at, 

75 last_modified_date=config.updated_at, 

76 )

78 @staticmethod 

79 def _check_data(data: dict) -> None: 

80 """Reject payloads Postgres can't store, or that exceed the size cap. 

81 

82 ``ensure_ascii=False`` matters: the default escapes every non-ASCII character 

83 to ``\\uXXXX``, which measures a 2-byte UTF-8 character as 6 and would reject a 

84 config at roughly a third of the advertised limit. 

85 """ 

86 # JSONB is text under the hood and Postgres rejects NUL in text. Pydantic accepts 

87 # it (it's legal JSON) and asyncpg then raises an untranslatable-character error 

88 # that no handler catches, so refuse it here as a clean 400. Checked on the parsed 

89 # structure, not the serialized form — json.dumps rewrites a real NUL to the 

90 # six-character text "\\u0000", which no substring check can tell apart from a 

91 # string that genuinely contains those characters. 

92 if _has_nul(data): 

93 raise BadRequest("Config data must not contain NUL characters") 

94 if len(json.dumps(data, ensure_ascii=False).encode("utf-8")) > MAX_DATA_BYTES: 

95 raise BadRequest(f"Config data exceeds the {MAX_DATA_BYTES // 1024} KB limit")

97 async def create(self, app: str, data: NewConfigDTO) -> ConfigDTO: 

98 self._check_data(data.data) 

99 _check_text(data.name, "name") 

100 _check_text(data.description, "description") 

101 _check_text(data.created_by, "createdBy") 

102 # Retry on the (astronomically unlikely) slug collision; a fresh slug per attempt. 

103 for attempt in range(_SLUG_MAX_RETRIES): 

104 config = Config( 

105 slug=_new_slug(), 

106 app=app, 

107 name=data.name, 

108 description=data.description, 

109 data=data.data, 

110 created_by=data.created_by, 

111 ) 

112 try: 

113 # SAVEPOINT, not a bare rollback: the session belongs to the caller 

114 # (see the class docstring), so unwinding the whole transaction here 

115 # would silently discard any write it made before calling us. 

116 async with self.session.begin_nested(): 

117 config = await self.repo.create(config) 

118 except IntegrityError: 

119 if attempt == _SLUG_MAX_RETRIES - 1: 

120 raise ConfigSlugConflict("Could not allocate a unique config slug") from None 

121 continue 

122 return self._to_dto(config) 

123 # Unreachable — the loop either returns or raises on the last attempt. 

124 raise ConfigSlugConflict("Could not allocate a unique config slug")

126 async def _get_for_app(self, app: str, slug: str) -> Config | None: 

127 """Load by slug, enforcing the app handshake: a mismatch reads as absent.""" 

128 config = await self.repo.get_by_slug(slug) 

129 if config is None or config.app != app: 

130 return None 

131 return config

133 async def get_by_slug(self, app: str, slug: str) -> ConfigDTO | None: 

134 config = await self._get_for_app(app, slug) 

135 return self._to_dto(config) if config else None

180 async def resolve_slug(self, slug: str) -> ConfigDTO | None: 

181 """Resolve by slug alone, bypassing the app handshake (debug only). 

182 

183 The normal read path is ``get_by_slug(app, slug)``, which 404s a slug that 

184 doesn't belong to ``app``. This answers "which app owns this slug?" instead, 

185 so it must not enforce the handshake. 

186 """ 

187 config = await self.repo.get_by_slug(slug) 

188 return self._to_dto(config) if config else None

190 async def update(self, app: str, slug: str, data: UpdateConfigDTO) -> ConfigDTO | None: 

191 config = await self._get_for_app(app, slug) 

192 if config is None: 

193 return None 

194 if data.data is not None: 

195 self._check_data(data.data) 

196 config.data = data.data 

197 if data.name is not None: 

198 _check_text(data.name, "name") 

199 config.name = data.name 

200 if data.description is not None: 

201 _check_text(data.description, "description") 

202 config.description = data.description 

203 config = await self.repo.update(config) 

204 return self._to_dto(config)

206 async def delete(self, app: str, slug: str) -> bool: 

207 config = await self._get_for_app(app, slug) 

208 if config is None: 

209 return False 

210 await self.repo.delete(config) 

211 return True
