38def _resolve_kind(kind: str | None, filename: str | None) -> Literal["edl", "ui"]: 

39 """Prefer the explicit ``kind`` form field; else infer from the filename suffix.""" 

40 if kind in ("edl", "ui"): 

41 return kind # type: ignore[return-value] 

42 if filename: 

43 guessed = _SUFFIX_KIND.get(Path(filename).suffix.lower()) 

44 if guessed is not None: 

45 return guessed 

46 raise UnsupportedInput("Could not determine screen kind; pass kind=edl|ui or upload a .edl/.ui file")

49async def _read_upload(file: UploadFile, max_bytes: int) -> bytes: 

50 # Reject on the declared size first so an oversized body is not fully buffered 

51 # into memory before the check. Content-Length can be absent or wrong, so the 

52 # post-read length check below stays as the authoritative backstop. 

53 if file.size is not None and file.size > max_bytes: 

54 raise UploadTooLarge(f"Upload is {file.size} bytes; limit is {max_bytes} bytes") 

55 data = await file.read() 

56 if len(data) > max_bytes: 

57 raise UploadTooLarge(f"Upload is {len(data)} bytes; limit is {max_bytes} bytes") 

58 return data

77@router.post("/conversions", dependencies=[Security(require_read_access)]) 

78async def create_conversion( 

79 service: ConversionServiceDep, 

80 config: ConverterConfigDep, 

81 fox_validate: FoxValidateDep, 

82 file: Annotated[UploadFile, File(description="A legacy .edl or .ui screen")], 

83 kind: Annotated[Literal["edl", "ui"] | None, Form()] = None, 

84) -> ScreenConversionResultDTO: 

85 """Convert one uploaded legacy screen to Canopy Screen IR (synchronous). 

86 

87 ``kind`` may be omitted when the filename carries a ``.edl``/``.ui`` suffix. 

88 The screen is validated against the Screen IR schema before it is returned. 

89 """ 

90 data = await _read_upload(file, config.max_upload_bytes) 

91 resolved = _resolve_kind(kind, file.filename) 

92 result = await run_in_threadpool(lambda: service.convert(data, kind=resolved, filename=file.filename)) 

93 # Seam #2: validate hoisted calc:// formulas via Fox; failures become warnings. 

94 formula_warnings = await validate_formulas(result.formulas, validate=fox_validate) 

95 return _to_dto(result, result.warnings + formula_warnings)

98@router.post("/conversions/validate", dependencies=[Security(require_read_access)]) 

99async def validate_conversion( 

100 service: ConversionServiceDep, 

101 config: ConverterConfigDep, 

102 fox_validate: FoxValidateDep, 

103 file: Annotated[UploadFile, File(description="A legacy .edl or .ui screen")], 

104 kind: Annotated[Literal["edl", "ui"] | None, Form()] = None, 

105) -> ValidateResultDTO: 

106 """Dry-run a conversion and return only diagnostics (the IR is discarded). 

107 

108 A pre-import check for the Screen Builder: ``ok`` is false when the screen 

109 cannot be parsed into valid IR; warnings and unknown widgets are always 

110 reported so the user sees what an import would surface. 

111 """ 

112 data = await _read_upload(file, config.max_upload_bytes) 

113 # Diagnostics endpoint: an undeterminable kind (UnsupportedInput from 

114 # _resolve_kind or service.convert) is a "not importable" result, not an 

115 # error envelope, so it must fold into ok=False like a parse failure. 

116 try: 

117 resolved = _resolve_kind(kind, file.filename) 

118 result = await run_in_threadpool(lambda: service.convert(data, kind=resolved, filename=file.filename)) 

119 except (UnsupportedInput, ConversionFailed) as exc: 

120 return ValidateResultDTO(ok=False, warnings=[exc.detail]) 

121 formula_warnings = await validate_formulas(result.formulas, validate=fox_validate) 

122 return ValidateResultDTO( 

123 ok=True, 

124 warnings=result.warnings + formula_warnings, 

125 unknown_widgets=[ 

126 UnknownWidgetDTO(original_class=u.original_class, count=u.count) for u in result.unknown_widgets 

127 ], 

128 )

131def _looks_like_archive(data: bytes) -> bool: 

132 """True for a zip (``PK\\x03\\x04``) or gzip (``\\x1f\\x8b``) payload.""" 

133 return data[:4] == b"PK\x03\x04" or data[:2] == b"\x1f\x8b"

136@router.post("/conversions/batch", dependencies=[Security(require_read_access)]) 

137async def create_batch_conversion( 

138 config: ConverterConfigDep, 

139 file: Annotated[UploadFile, File(description="A .zip or .tar.gz of legacy .edl/.ui screens")], 

140) -> JobCreatedDTO: 

141 """Convert a folder of legacy screens asynchronously. 

142 

143 Accepts one archive (``.zip``/``.tar.gz``), stages it, and enqueues a job. 

144 Poll ``GET /jobs/v1/{jobId}``: its ``result`` is a ``BatchManifestDTO`` whose 

145 ``succeeded`` rows carry an ``index``; fetch each screen's IR from 

146 ``GET /conversions/batch/{jobId}/screens/{index}``. 

147 """ 

148 # Lazy import: keeps the converter loadable without Jobs and routes the 

149 # cross-module call through the public gateway only (mirrors get_fox_validate). 

150 from canopy.jobs import gateway as jobs_gateway 

151 from canopy.converter.tasks import input_key 

152 

153 data = await _read_upload(file, config.max_archive_bytes) 

154 if not _looks_like_archive(data): 

155 raise UnsupportedInput("Batch upload must be a .zip or .tar.gz archive") 

156 

157 token = uuid4().hex 

158 pool = await jobs_gateway.get_pool() 

159 await pool.set(input_key(token), data, ex=config.batch_result_ttl_s) 

160 job_id = await jobs_gateway.enqueue("convert_batch", token) 

161 return JobCreatedDTO(job_id=job_id, status_url=f"/jobs/v1/{job_id}")

164@router.get("/conversions/batch/{job_id}/screens/{index}", dependencies=[Security(require_read_access)]) 

165async def get_batch_screen(job_id: str, index: int) -> BatchScreenDTO: 

166 """Fetch one converted screen's IR from a finished batch by its manifest ``index``. 

167 

168 Diagnostics (warnings, unknown widgets) live in the batch manifest row; this 

169 returns just the Screen IR document the editor loads. 

170 """ 

171 from canopy.jobs import gateway as jobs_gateway 

172 from canopy.converter.tasks import result_key 

173 

174 pool = await jobs_gateway.get_pool() 

175 raw = await pool.get(result_key(job_id, index)) 

176 if raw is None: 

177 raise BatchResultNotFound(f"No batch screen for job {job_id!r} index {index} (expired or unknown)") 

178 return BatchScreenDTO(screen=json.loads(raw))
