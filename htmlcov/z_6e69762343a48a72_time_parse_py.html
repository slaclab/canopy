1"""Parse the `from` / `to` query params into UTC nanoseconds. 

2 

3Accepts ISO-8601 (with optional `Z` suffix) or epoch-nanoseconds (as a string 

4or int). Used by the data, exports, and (eventually) search endpoints. 

5 

6Also holds Canopy's one rule for timezone-less datetimes (``to_utc``: a naive 

7value is read as UTC) and ``DateTimeBound``, the query-parameter type for 

8endpoints that take a datetime range bound. Declare it as 

9``Annotated[DateTimeBound, Query(...)] = None``: FastAPI drops the validator 

10when the ``Query`` is the parameter default instead. 

11""" 

12 

13import re 

14import time 

15from typing import Annotated 

16from datetime import UTC, datetime 

17 

18from pydantic import BeforeValidator 

19 

20from canopy.core.errors import BadRequest 

21 

22_DATE_ONLY = re.compile(r"\d{4}-\d{2}-\d{2}") 

23 

24# Epoch-nanoseconds must fit a signed 64-bit range and be non-negative. Beyond 

25# this, downstream datetime.fromtimestamp / int64 conversions raise OverflowError 

26# / OSError and surface as 500s; reject at the boundary as a clean 400 instead. 

27_MIN_NS = 0 

28_MAX_NS = 9_223_372_036_854_775_807 # int64 max (~year 2262) 

29 

30 

31def _bounded(ns: int, original: object) -> int: 

32 if ns < _MIN_NS or ns > _MAX_NS: 

33 raise BadRequest(f"Time out of range: {original!r}") 

34 return ns 

35

37def parse_time(value: str | int | None) -> int | None: 

38 """Parse an input value into UTC nanoseconds. 

39 

40 `None` → `None` (caller decides default). Empty string → `None` likewise. 

41 Raises `BadRequest` on malformed or out-of-range input. 

42 """ 

43 if value is None or value == "": 

44 return None 

45 if isinstance(value, int): 

46 return _bounded(value, value) 

47 try: 

48 return _bounded(int(value), value) 

49 except ValueError: 

50 pass 

51 s = value[:-1] + "+00:00" if value.endswith("Z") else value 

52 try: 

53 dt = datetime.fromisoformat(s) 

54 except ValueError as exc: 

55 raise BadRequest(f"Invalid time value: {value!r}") from exc 

56 try: 

57 dt = to_utc(dt) 

58 secs = int(dt.timestamp()) 

59 except (OverflowError, OSError) as exc: 

60 raise BadRequest(f"Time out of range: {value!r}") from exc 

61 return _bounded(secs * 1_000_000_000 + dt.microsecond * 1000, value)

64def to_utc(value: datetime) -> datetime: 

65 """Return ``value`` as a timezone-aware UTC datetime. 

66 

67 Canopy reads a datetime that arrives without an offset as UTC, and converts 

68 one that carries an offset to UTC, so it can be compared against 

69 ``timestamptz`` columns and nanosecond timestamps alike. 

70 

71 Parameters 

72 ---------- 

73 value : datetime 

74 A naive or timezone-aware datetime. 

75 

76 Returns 

77 ------- 

78 datetime 

79 The same instant with ``tzinfo`` set to UTC. 

80 """ 

81 if value.tzinfo is None: 

82 return value.replace(tzinfo=UTC) 

83 return value.astimezone(UTC)

86def reject_date_only(value: object) -> object: 

87 """Refuse a bare ``YYYY-MM-DD`` where a date and time are expected. 

88 

89 Pydantic reads a date-only string as midnight, which silently turns an 

90 inclusive upper bound such as ``end=2026-01-31`` into one that excludes that 

91 whole day. Requiring the time makes the caller state which instant it means. 

92 

93 Parameters 

94 ---------- 

95 value : object 

96 The raw query value, before datetime parsing. 

97 

98 Returns 

99 ------- 

100 object 

101 ``value`` unchanged when it is not a bare date. 

102 

103 Raises 

104 ------ 

105 ValueError 

106 When ``value`` is a string holding only a date. 

107 """ 

108 if isinstance(value, str) and _DATE_ONLY.fullmatch(value.strip()): 

109 raise ValueError("expected a date and time such as 2026-01-31T23:59:59Z, not a bare date") 

110 return value

113DateTimeBound = Annotated[datetime | None, BeforeValidator(reject_date_only)]

120def parse_bin_size_to_ns(value: str) -> int: 

121 """Parse a `binSize` query value into nanoseconds. 

122 

123 v1 accepts integer nanoseconds (as a string). ISO-8601 durations are not 

124 supported yet — the frontend can compute ns. Raises `BadRequest` on malformed input. 

125 """ 

126 try: 

127 ns = int(value) 

128 except (TypeError, ValueError) as exc: 

129 raise BadRequest(f"Invalid binSize: {value!r}; expected nanoseconds as integer") from exc 

130 if ns < 1: 

131 raise BadRequest("binSize must be >= 1 ns") 

132 return ns
