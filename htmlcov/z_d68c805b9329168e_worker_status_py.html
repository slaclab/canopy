1"""The worker's own view of its restore hooks, handed to the API through Redis. 

2 

3Restores run in ``canopy-worker``, so the hooks that matter are the ones the worker 

4resolved from its environment. The worker publishes ``describe_restore_hooks()`` at 

5startup and the API's status endpoint reads it back. That makes a 

6``CANOPY_SQUIRREL__RESTORE_HOOKS`` that reached the API container but not the worker 

7visible, instead of reporting the API's view as if it were the worker's. 

8 

9The report has no expiry: it is replaced on every worker start and carries the time 

10and host it was written from, so a stale one is recognisable rather than vanishing. 

11""" 

12 

13from __future__ import annotations 

14 

15import json 

16import socket 

17from typing import Any 

18from datetime import UTC, datetime 

19 

20from redis.asyncio import Redis 

21 

22from canopy.squirrel.restore_hooks.registry import describe_restore_hooks 

23 

24WORKER_STATUS_KEY = "canopy:squirrel:restore_hooks:worker" 

25 

26 

27async def publish_worker_status(redis: Redis) -> dict[str, Any]: 

28 """Record this process's restore-hook resolution as the worker's report. 

29 

30 Parameters 

31 ---------- 

32 redis : Redis 

33 The worker's Redis connection (Arq's ``ctx["redis"]``). 

34 

35 Returns 

36 ------- 

37 dict[str, Any] 

38 The report written: ``describe_restore_hooks()`` plus ``reported_at`` (ISO-8601, 

39 UTC) and ``host``. 

40 """ 

41 report = { 

42 **describe_restore_hooks(), 

43 "reported_at": datetime.now(UTC).isoformat(), 

44 "host": socket.gethostname(), 

45 } 

46 await redis.set(WORKER_STATUS_KEY, json.dumps(report)) 

47 return report 

48 

49 

50async def read_worker_status(redis: Redis) -> dict[str, Any] | None: 

51 """Read the report the worker last published. 

52 

53 Parameters 

54 ---------- 

55 redis : Redis 

56 Any connection to the Redis the worker uses. 

57 

58 Returns 

59 ------- 

60 dict[str, Any] | None 

61 The report as published, or ``None`` when no worker has published one. 

62 """ 

63 raw = await redis.get(WORKER_STATUS_KEY) 

64 return None if raw is None else json.loads(raw)
