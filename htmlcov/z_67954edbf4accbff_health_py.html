36class WorkerReport:

52async def _worker_report() -> WorkerReport: 

53 """Fetch the worker's published restore-hook report from Redis. 

54 

55 Returns 

56 ------- 

57 WorkerReport 

58 The worker's view, or why it could not be read. 

59 """ 

60 try: 

61 async with asyncio.timeout(WORKER_STATUS_TIMEOUT_S): 

62 report = await read_worker_status(await jobs_gateway.get_pool()) 

63 except (TimeoutError, OSError, RedisError) as exc: 

64 return WorkerReport(error=f"Could not read the worker's report from Redis: {str(exc) or type(exc).__name__}") 

65 return WorkerReport(view=None if report is None else WorkerRestoreHooksViewDTO(**report))

68@router.get("/restore-hooks", dependencies=[Security(require_read_access)]) 

69async def restore_hooks_status() -> RestoreHooksStatusDTO: 

70 """Report which restore hooks the worker and this API process resolved. 

71 

72 Restore hooks send extra per-PV writes during a snapshot restore (magnet trims, 

73 for instance) and are selected with ``CANOPY_SQUIRREL__RESTORE_HOOKS``. A variable 

74 that never reaches the worker, or a mistyped provider key, leaves the hook silently 

75 doing nothing — this endpoint makes that visible without running a restore. 

76 

77 Restores run in ``canopy-worker``, so its view comes from the report it publishes 

78 to Redis at startup. The API's own view is computed here from configuration 

79 only: no hook is built or cached. 

80 

81 Returns 

82 ------- 

83 RestoreHooksStatusDTO 

84 Both views, whether they load the same hooks, and every available provider. 

85 """ 

86 api = describe_restore_hooks() 

87 worker = await _worker_report() 

88 return RestoreHooksStatusDTO( 

89 available=api["available"], 

90 api=RestoreHooksViewDTO(**api), 

91 worker=worker.view, 

92 worker_error=worker.error, 

93 matches_api=None if worker.view is None else worker.view.loaded == api["loaded"], 

94 )

100def _from_unix(value: float | None) -> datetime | None: 

101 """``value`` as a UTC datetime; ``None`` for no value, 0 (a stepped-down leader) or one out of range.""" 

102 if not value: 

103 return None 

104 try: 

105 return datetime.fromtimestamp(value, tz=UTC) 

106 except (OverflowError, OSError, ValueError): 

107 return None

115@router.get("/pv-cache", dependencies=[Security(require_read_access)]) 

116async def pv_cache_status() -> PVCacheStatusDTO: 

117 """Report the worker-side PV cache that cached snapshots (``useCache=true``) read. 

118 

119 The cache is written by whichever ``canopy-worker`` holds the leader lease, which 

120 publishes a report to Redis on every flush. A worker that never got 

121 ``CANOPY_SQUIRREL__PV_CACHE__ENABLED``, or one that stopped, leaves cached 

122 snapshots silently reading live; this endpoint shows which it is. 

123 

124 Returns 

125 ------- 

126 PVCacheStatusDTO 

127 The leader's last report, its heartbeat age, and the overall status. 

128 """ 

129 cfg = get_squirrel_config().pv_cache 

130 try: 

131 async with asyncio.timeout(PV_CACHE_STATUS_TIMEOUT_S): 

132 pool = await jobs_gateway.get_pool() 

133 cached = await pool.hlen(VALUES_KEY) 

134 try: 

135 report, unreadable = await read_status(pool), None 

136 except UnreadableStatus as exc: 

137 report, unreadable = None, exc 

138 except (TimeoutError, OSError, RedisError) as exc: 

139 return PVCacheStatusDTO( 

140 status="down", 

141 enabled=cfg.enabled, 

142 error=f"Could not read the PV cache report from Redis: {str(exc) or type(exc).__name__}", 

143 ) 

144 

145 now = time.time() 

146 if unreadable is not None: 

147 error = _sentence(str(unreadable)) 

148 elif report is None: 

149 error = "No worker has reported a PV cache; is CANOPY_SQUIRREL__PV_CACHE__ENABLED set on canopy-worker?" 

150 else: 

151 # The same check snapshot create applies, so "ok" here means snapshots use the cache. 

152 reason = untrusted_reason(report, now, cfg.heartbeat_max_age_s) 

153 error = None if reason is None else _sentence(reason) 

154 

155 # Off, not broken, only when this process has the cache off too and no worker is 

156 # leading it: no report, or one a leader left on stepping down. The flag is often 

157 # set on the worker alone, and a leader that crashed never writes the step-down, 

158 # so its stale report stays "down" here until it expires. 

159 switched_off = not cfg.enabled and unreadable is None and (report is None or not report.heartbeat_at) 

160 if error is not None and switched_off: 

161 status, error = "disabled", None 

162 elif error is not None: 

163 status = "down" 

164 elif not report.warm or ( 

165 report.monitored is not None and report.connected is not None and report.connected < report.monitored 

166 ): 

167 status = "degraded" 

168 else: 

169 status = "ok" 

170 if report is None: 

171 return PVCacheStatusDTO(status=status, enabled=cfg.enabled, cached=cached, error=error) 

172 return PVCacheStatusDTO( 

173 status=status, 

174 enabled=cfg.enabled, 

175 host=report.host, 

176 instance=report.instance, 

177 leader_since=_from_unix(report.leader_since), 

178 heartbeat_at=_from_unix(report.heartbeat_at), 

179 heartbeat_age_s=round(now - report.heartbeat_at, 3) if report.heartbeat_at else None, 

180 configured=report.configured, 

181 monitored=report.monitored, 

182 connected=report.connected, 

183 seen=report.seen, 

184 warm=report.warm, 

185 cached=cached, 

186 error=error, 

187 )
