1"""Config store endpoints. 

2 

3Endpoints are app-scoped in the path (``/configs/{app}/...``). Since slugs are 

4globally unique, ``app`` is not needed to *find* a config — it's an assertion: the 

5service treats a config whose ``app`` doesn't match the path as "not found". This 

6handshake stops a consumer from accidentally loading another app's config. It is a 

7correctness guard, not a security boundary (any valid key can read any app by 

8naming it). Auth is the standard ``X-API-Key`` read/write scopes. 

9 

10``app`` is constrained to ``[a-z0-9_-]{1,64}`` — lowercase only, so ``Trace`` and 

11``my.app`` are rejected rather than silently treated as apps of their own. The 

12routes that write or filter on it (create, list) reject a malformed name with 

13``invalid_app_name``; the item routes treat it as an assertion and 404 instead. 

14""" 

15 

16from typing import Annotated 

17 

18from fastapi import Path, Query, Security, APIRouter 

19 

20from canopy.core.errors import BadRequest 

21from canopy.core.security import require_read_access, require_write_access 

22from canopy.magpie.errors import ConfigNotFound, InvalidAppName 

23from canopy.magpie.dependencies import ConfigServiceDep 

24from canopy.contracts.pagination import PagedResult 

25from canopy.magpie.schemas.config import ConfigDTO, NewConfigDTO, UpdateConfigDTO 

26from canopy.magpie.api.v1._validation import APP_NAME_RULE, is_app, is_slug, is_uuid 

27 

28router = APIRouter(prefix="/configs", tags=["Configs"]) 

29 

30# Documents the charset in the OpenAPI schema so the rule is discoverable before a 

31# client trips over a 400. 

32AppPath = Annotated[str, Path(description=f"Owning app ({APP_NAME_RULE})")] 

33 

34 

35def _validate_app_name(app: str) -> None: 

36 """Reject a malformed ``app`` name. Format only — not a DB existence check. 

37 

38 Used by the routes that write or filter on ``app`` (create, list). The item 

39 routes deliberately skip it and let the app handshake 404 a mismatch instead. 

40 """ 

41 if not is_app(app): 

42 raise InvalidAppName(f"Invalid app name '{app}' ({APP_NAME_RULE})") 

43 

44 

45@router.post("/{app}", dependencies=[Security(require_write_access)]) 

46async def create_config(app: AppPath, data: NewConfigDTO, service: ConfigServiceDep) -> ConfigDTO: 

47 """Create a config under ``app``.""" 

48 _validate_app_name(app) 

49 return await service.create(app, data) 

50 

51 

52@router.get("/{app}", dependencies=[Security(require_read_access)]) 

53async def list_configs( 

54 app: AppPath, 

55 service: ConfigServiceDep, 

56 search: str | None = Query(None, max_length=255, description="Case-insensitive contains over name/description"), 

57 pageSize: int = Query(100, ge=1, le=1000), 

58 continuationToken: str | None = Query(None), 

59) -> PagedResult[ConfigDTO]: 

60 """List an app's configs (keyset-paginated).""" 

61 _validate_app_name(app) 

62 # The token is a row id; letting a malformed one through would reach asyncpg as a 

63 # bad UUID comparison and surface as a 500 rather than a 400. 

64 if continuationToken is not None and not is_uuid(continuationToken): 

65 raise BadRequest("Invalid continuationToken") 

66 return await service.list_by_app( 

67 app=app, 

68 search=search, 

69 page_size=pageSize, 

70 continuation_token=continuationToken, 

71 ) 

72 

73 

74@router.get("/{app}/{slug}", dependencies=[Security(require_read_access)]) 

75async def get_config(app: AppPath, slug: str, service: ConfigServiceDep) -> ConfigDTO: 

76 """Resolve a config by slug (the shareable-link endpoint).""" 

77 if not is_slug(slug): 

78 raise ConfigNotFound(f"Config '{slug}' not found") 

79 config = await service.get_by_slug(app, slug) 

80 if config is None: 

81 raise ConfigNotFound(f"Config '{slug}' not found") 

82 return config 

83 

84 

85@router.put("/{app}/{slug}", dependencies=[Security(require_write_access)]) 

86async def update_config(app: AppPath, slug: str, data: UpdateConfigDTO, service: ConfigServiceDep) -> ConfigDTO: 

87 """Update a config's name/description/data (app and slug are immutable).""" 

88 if not is_slug(slug): 

89 raise ConfigNotFound(f"Config '{slug}' not found") 

90 config = await service.update(app, slug, data) 

91 if config is None: 

92 raise ConfigNotFound(f"Config '{slug}' not found") 

93 return config 

94 

95 

96@router.delete("/{app}/{slug}", dependencies=[Security(require_write_access)]) 

97async def delete_config(app: AppPath, slug: str, service: ConfigServiceDep) -> bool: 

98 """Delete a config.""" 

99 if not is_slug(slug): 

100 raise ConfigNotFound(f"Config '{slug}' not found") 

101 if not await service.delete(app, slug): 

102 raise ConfigNotFound(f"Config '{slug}' not found") 

103 return True
