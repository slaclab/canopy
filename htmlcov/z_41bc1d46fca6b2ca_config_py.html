1"""E-log configuration. 

2 

3Sourced from env vars with prefix ``CANOPY_ELOG__`` and nested delimiter 

4``__``. Each client app gets its own elog-plus application token, so the 

5browser and desktop clients never hold one:: 

6 

7 CANOPY_ELOG__PROVIDER=elog_plus 

8 CANOPY_ELOG__ELOG_PLUS__BASE_URL=https://elog.example.org/api/elog-apptoken 

9 CANOPY_ELOG__APPS__SQUIRREL__TOKEN=<elog-plus application token> 

10 CANOPY_ELOG__APPS__SQUIRREL__DEFAULT_LOGBOOK_IDS=["<logbook id>"] 

11 CANOPY_ELOG__APPS__TRACE__TOKEN=<elog-plus application token> 

12 

13The app name is the ``{app}`` path segment of every route; env var names are 

14case-insensitive, so ``APPS__SQUIRREL__`` lands in ``apps["squirrel"]``. 

15""" 

16 

17from pydantic import Field, BaseModel, SecretStr, ValidationError, field_validator 

18from pydantic_settings import BaseSettings, SettingsError, SettingsConfigDict 

19 

20 

21class ElogPlusConfig(BaseModel): 

22 """Settings for the ``elog_plus`` provider (https://github.com/slaclab/elog-plus). 

23 

24 ``base_url`` must be elog-plus' application-token ingress. At SLAC that is 

25 ``/api/elog-apptoken``, not ``/api/elog``: the latter sits behind Vouch, 

26 which strips the token header before it reaches elog-plus. 

27 """ 

28 

29 base_url: str = "" 

30 # Matches ELOG_PLUS_AUTH_HEADER on the elog-plus service. 

31 auth_header: str = "x-vouch-idp-accesstoken" 

32 

33 

34class ElogAppConfig(BaseModel): 

35 """One client app (``squirrel``, ``trace``, ...) and its elog-plus identity.""" 

36 

37 token: SecretStr | None = None 

38 # Logbooks a client preselects in its post dialog. JSON list in env. 

39 default_logbook_ids: list[str] = Field(default_factory=list) 

40 

41 @field_validator("token") 

42 @classmethod 

43 def _strip_token(cls, value: SecretStr | None) -> SecretStr | None: 

44 # A token read from a secret file often ends in a newline, which no HTTP 

45 # header value may carry; every request would then fail. 

46 if value is None: 

47 return None 

48 return SecretStr(value.get_secret_value().strip())

51class ElogConfig(BaseSettings): 

52 model_config = SettingsConfigDict( 

53 env_prefix="CANOPY_ELOG__", 

54 env_nested_delimiter="__", 

55 env_file=".env", 

56 extra="ignore", 

57 ) 

58 

59 # Empty disables the module: every app's config reports enabled=false. 

60 provider: str = "" 

61 elog_plus: ElogPlusConfig = ElogPlusConfig() 

62 apps: dict[str, ElogAppConfig] = {} 

63 # Outbound HTTP(S) proxy for e-log calls. Process proxy env vars are 

64 # deliberately ignored (the client is built with trust_env=False), but 

65 # SSL_CERT_FILE and SSL_CERT_DIR are still honored for a custom CA bundle. 

66 proxy_url: str | None = None 

67 timeout_s: float = 15.0 

68 # Reject attachment uploads larger than this (bytes) with 413. 

69 max_attachment_bytes: int = 20 * 1024 * 1024 

70 

71 @field_validator("provider") 

72 @classmethod 

73 def _strip_provider(cls, value: str) -> str: 

74 return value.strip() 

75 

76 @field_validator("proxy_url") 

77 @classmethod 

78 def _empty_proxy_is_none(cls, value: str | None) -> str | None: 

79 if value is None: 

80 return None 

81 return value.strip() or None

84def get_elog_config() -> ElogConfig: 

85 return ElogConfig()

88def describe_settings_error(exc: ValidationError | SettingsError) -> str: 

89 """Why the ``CANOPY_ELOG__*`` settings failed to parse, safe to show any caller. 

90 

91 Names the offending variables but never their values: pydantic's own 

92 message echoes the input, which may sit next to a token in a deploy secret. 

93 """ 

94 if isinstance(exc, ValidationError): 

95 problems = [ 

96 f"{_env_name(error['loc'])}: {error['msg']}" 

97 for error in exc.errors(include_input=False, include_url=False, include_context=False) 

98 ] 

99 return "Invalid e-log settings: " + "; ".join(problems) 

100 # SettingsError names the field and source only (e.g. a DEFAULT_LOGBOOK_IDS that isn't JSON). 

101 return f"Invalid e-log settings: {exc}" 

102 

103 

104def _env_name(loc: tuple) -> str: 

105 return "CANOPY_ELOG__" + "__".join(str(part).upper() for part in loc)
