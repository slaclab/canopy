1"""`KeyRegistry` implementations for persistent keys. 

2 

3Three small pieces, composed in ``canopy.main``'s lifespan: 

4 

5 ChainedKeyRegistry( 

6 EnvKeyRegistry(settings.api_keys), # statically stable floor 

7 CachingKeyRegistry(DbKeyRegistry()), # revocable, DB-backed 

8 ) 

9 

10``DbKeyRegistry`` only knows how to look a token up. ``CachingKeyRegistry`` owns 

11the TTL, the negative cache, and the serve-stale behaviour, so that policy is 

12testable without a database and reusable over any inner registry. 

13"""

35# What "the store is unreachable" actually looks like. asyncpg surfaces a refused 

36# connection as a bare OSError raised by asyncio's `create_connection`; SQLAlchemy 

37# never wraps it. TimeoutError subclasses OSError, so connect timeouts are covered 

38# too. OperationalError/InterfaceError/SATimeoutError are SQLAlchemy's own 

39# connection-level failures (lost connection, pool exhaustion) — not the broader 

40# `SQLAlchemyError`, which also covers schema/query bugs that must not be reported 

41# as a database outage and served stale for up to an hour. 

42STORE_ERRORS = (OperationalError, InterfaceError, SATimeoutError, OSError)

58class DbKeyRegistry: 

59 """Resolves a token against the ``core.api_key`` table. 

60 

61 Opens its own short-lived session rather than taking the request-scoped one: 

62 this runs inside a `Security` dependency on every route, and must not hold a 

63 transaction open for the life of a request. 

64 

65 Also stamps ``last_used_at`` on the way through. That write is naturally 

66 throttled by the caching layer in front — a given token only reaches here once 

67 per cache TTL — so it needs no throttling machinery of its own. 

68 """ 

69 

70 async def get(self, token: str) -> APIKey | None: 

71 token_hash = hash_token(token) 

72 async with session_scope() as session: 

73 repo = ApiKeyRepository(session) 

74 row = await repo.get_by_token_hash(token_hash, active_only=True) 

75 if row is None: 

76 return None 

77 key = _to_api_key(row) 

78 try: 

79 await repo.touch_last_used(row.id) 

80 await session.commit() 

81 except STORE_ERRORS: 

82 # Attribution is a nicety; never fail an otherwise valid request for it. 

83 logger.warning("Could not stamp last_used_at for API key id=%s", row.id) 

84 await session.rollback() 

85 return key

94class CachingKeyRegistry: 

95 """TTL cache in front of another registry, with bounded serve-stale. 

96 

97 Three behaviours worth stating outright, because they are the auth contract: 

98 

99 * **Revocation latency equals ``ttl_s``, except locally.** The process that runs 

100 the deactivate calls ``invalidate_key_id`` and stops accepting the key at once; 

101 other replicas keep it working until their cached entry expires. 

102 * **The store going down does not take auth down.** If the inner registry 

103 raises, a cached key is served up to ``max_stale_s`` past its fetch. Canopy 

104 has ten mounted sub-apps of which seven need no database at all; a Postgres 

105 outage must not black out live PV streaming. Past ``max_stale_s`` the entry 

106 is refused, so a revoked key cannot survive an arbitrarily long outage. 

107 * **Negatives are never served stale.** An unknown token during an outage 

108 fails closed, with a distinguishable 503 rather than a misleading 401. 

109 """ 

110 

111 def __init__( 

112 self, 

113 inner: KeyRegistry, 

114 ttl_s: float = 60.0, 

115 negative_ttl_s: float = 5.0, 

116 max_stale_s: float = 3600.0, 

117 max_entries: int = 10_000, 

118 store_error_backoff_s: float = 5.0, 

119 ) -> None: 

120 self._inner = inner 

121 self._ttl_s = ttl_s 

122 self._negative_ttl_s = negative_ttl_s 

123 self._max_stale_s = max_stale_s 

124 self._max_entries = max_entries 

125 self._store_error_backoff_s = store_error_backoff_s 

126 self._cache: dict[str, _CacheEntry] = {} 

127 # Set once the inner registry raises, so the *next* request in the backoff 

128 # window serves stale immediately instead of paying the same failed round 

129 # trip (pool timeout, connect timeout) all over again. 

130 self._store_down_until: float = 0.0 

131 # One in-flight fetch per token: N concurrent requests for the same cold or 

132 # expired token would otherwise each open a session and hit the row, which 

133 # is how a reconnect storm exhausts the pool. 

134 self._inflight: dict[str, asyncio.Task[APIKey | None]] = {} 

135 

136 async def get(self, token: str) -> APIKey | None: 

137 # Minted tokens have a fixed shape; anything else (a scanner, a misconfigured 

138 # client) can be rejected before it costs a session + SELECT on the shared 

139 # pool, and without ever touching the store, so it can't 503 during an outage. 

140 if not token.startswith(TOKEN_PREFIX) or len(token) != MINTED_TOKEN_LEN: 

141 return None 

142 

143 # Keyed by hash so a long-lived process dict never holds plaintext tokens. 

144 cache_key = hash_token(token) 

145 now = monotonic() 

146 

147 entry = self._cache.get(cache_key) 

148 if entry is not None: 

149 ttl = self._ttl_s if entry.key is not None else self._negative_ttl_s 

150 if now - entry.fetched_at < ttl: 

151 return entry.key 

152 

153 if now < self._store_down_until: 

154 return self._serve_stale(entry, now, exc=None) 

155 

156 task = self._inflight.get(cache_key) 

157 if task is not None: 

158 return await task 

159 

160 task = self._inflight[cache_key] = asyncio.ensure_future(self._fetch(token, cache_key, entry, now)) 

161 try: 

162 return await task 

163 finally: 

164 self._inflight.pop(cache_key, None) 

165 

166 async def _fetch(self, token: str, cache_key: str, entry: _CacheEntry | None, now: float) -> APIKey | None: 

167 try: 

168 key = await self._inner.get(token) 

169 except STORE_ERRORS as exc: 

170 self._store_down_until = monotonic() + self._store_error_backoff_s 

171 return self._serve_stale(entry, now, exc) 

172 

173 self._store(cache_key, key, now) 

174 return key 

175 

176 def _serve_stale(self, entry: _CacheEntry | None, now: float, exc: Exception | None) -> APIKey: 

177 if entry is not None and entry.key is not None and now - entry.fetched_at <= self._max_stale_s: 

178 logger.warning( 

179 "API key store unavailable; serving cached key for app_name=%s (%.0fs stale)", 

180 entry.key.name, 

181 now - entry.fetched_at, 

182 ) 

183 return entry.key 

184 if exc is not None: 

185 logger.error("API key store unavailable with no usable cached entry: %s", exc) 

186 raise KeyStoreUnavailable("API key store is unavailable") from exc 

187 

188 def _store(self, cache_key: str, key: APIKey | None, now: float) -> None: 

189 # Positives are bounded by the number of real keys, so they are always 

190 # cached. Negatives are attacker-controlled: a flood of invented tokens 

191 # would otherwise grow this dict without limit, so once the cache is full 

192 # they go uncached and simply pay the lookup each time. 

193 if key is None and len(self._cache) >= self._max_entries: 

194 self._prune(now) 

195 if len(self._cache) >= self._max_entries: 

196 return 

197 self._cache[cache_key] = _CacheEntry(key=key, fetched_at=now) 

198 

199 def _prune(self, now: float) -> None: 

200 """Drop entries that can no longer be used for anything — negatives past 

201 their TTL, positives past the stale horizon.""" 

202 for cache_key, entry in list(self._cache.items()): 

203 horizon = self._max_stale_s if entry.key is not None else self._negative_ttl_s 

204 if now - entry.fetched_at > horizon: 

205 del self._cache[cache_key] 

206 

207 def invalidate_key_id(self, key_id: str) -> None: 

208 """Drop a key's cached entry, by id, so a revocation takes effect 

209 immediately in this process rather than waiting out ``ttl_s``. 

210 

211 Keyed by id rather than token: the deactivate paths (the DELETE route, the 

212 CLI) only ever have the row's id, never the plaintext token. 

213 """ 

214 for cache_key, entry in list(self._cache.items()): 

215 if entry.key is not None and entry.key.id == key_id: 

216 del self._cache[cache_key]

219class ChainedKeyRegistry: 

220 """Try each registry in order; first match wins. 

221 

222 Env keys are checked before the database so the highest-volume credential in the 

223 system — the read key baked into every display — never makes a database round 

224 trip at all, and is unaffected by database latency as well as outages. 

225 

226 An unavailable layer does not end the search: it is remembered and the remaining 

227 layers are still tried, so a layer that *can* answer still does. Only if nothing 

228 matched and something was down does the outage surface — reporting "invalid key" 

229 for a token that could not actually be checked would send an operator hunting for 

230 the wrong problem. This keeps the ordering a performance decision rather than a 

231 correctness one: reordering the chain cannot silently take auth down during an 

232 outage. 

233 """ 

234 

235 def __init__(self, *registries: KeyRegistry) -> None: 

236 self._registries = registries 

237 

238 def invalidate_key_id(self, key_id: str) -> None: 

239 """Forward to every inner registry that caches (currently only 

240 `CachingKeyRegistry`) so a revocation drops out of the chain immediately.""" 

241 for registry in self._registries: 

242 invalidate = getattr(registry, "invalidate_key_id", None) 

243 if invalidate is not None: 

244 invalidate(key_id) 

245 

246 async def get(self, token: str) -> APIKey | None: 

247 unavailable: KeyStoreUnavailable | None = None 

248 for registry in self._registries: 

249 try: 

250 key = await registry.get(token) 

251 except KeyStoreUnavailable as exc: 

252 unavailable = exc 

253 continue 

254 if key is not None: 

255 return key 

256 if unavailable is not None: 

257 raise unavailable 

258 return None
