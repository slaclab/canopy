1"""Facility hierarchy endpoints. 

2 

3Writes require the facility wildcard, not just ``admin`` — stricter than key 

4management (see ``core/keys/routes.py``), because reparenting or deleting a node 

5can widen what every *existing* grant on an ancestor resolves to, not just the 

6grant the acting key is minting. A facility-scoped admin delegating a narrower 

7*key* grant is a separate check, in ``core.keys.service``. ``GET`` accepts 

8``read`` or ``admin`` — facility *existence* isn't a secret, and an admin-only 

9provisioning key needs to see what it just defined without also being handed an 

10unrelated data-plane scope — which is why ``Security`` is per-route rather than 

11on the router. 

12 

13Write routes are keyed by slug — it's immutable, and it's how a facility is spelled 

14everywhere else (``?facility=``, key grants). ``FacilityService`` takes ids, so each 

15route resolves the slug first. 

16""" 

17 

18from fastapi import Security, APIRouter 

19 

20from canopy.core.errors import FacilityNotFound 

21from canopy.core.security import ( 

22 require_read_or_admin_access, 

23 require_wildcard_facility_access, 

24) 

25from canopy.core.facilities.schemas import FacilityDTO, FacilityCreate, FacilityUpdate 

26from canopy.core.facilities.dependencies import FacilityServiceDep 

27 

28router = APIRouter(prefix="/v1/facilities", tags=["Facilities"]) 

29 

30 

31@router.get("", dependencies=[Security(require_read_or_admin_access)]) 

32async def list_facilities(service: FacilityServiceDep) -> list[FacilityDTO]: 

33 """Full facility hierarchy, flat with ``parent_id`` — the client assembles the tree.""" 

34 return await service.get_all() 

35 

36 

37@router.post("", dependencies=[Security(require_wildcard_facility_access)]) 

38async def create_facility(data: FacilityCreate, service: FacilityServiceDep) -> FacilityDTO: 

39 """Define a facility. ``slug`` is immutable once created — key grants and 

40 ``?facility=`` filters reference it, so renaming would silently invalidate them.""" 

41 return await service.create(data) 

42 

43 

44@router.put("/{facility_slug}", dependencies=[Security(require_wildcard_facility_access)]) 

45async def update_facility(facility_slug: str, data: FacilityUpdate, service: FacilityServiceDep) -> FacilityDTO: 

46 """Update a facility's name, description, or parent. ``slug`` is not editable.""" 

47 facility = await service.get_by_slug(facility_slug) 

48 if not facility: 

49 raise FacilityNotFound(f"Facility '{facility_slug}' not found") 

50 return await service.update(facility.id, data) 

51 

52 

53@router.delete("/{facility_slug}", dependencies=[Security(require_wildcard_facility_access)]) 

54async def delete_facility(facility_slug: str, service: FacilityServiceDep) -> dict[str, bool]: 

55 """Delete a facility. Refused while it has children, or while any module's data or 

56 any active key's grant still references it.""" 

57 facility = await service.get_by_slug(facility_slug) 

58 if not facility: 

59 raise FacilityNotFound(f"Facility '{facility_slug}' not found") 

60 await service.delete(facility.id) 

61 return {"deleted": True}
