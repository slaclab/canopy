61def normalize_severity(value: int | None) -> AlarmSeverity: 

62 """Map an EPICS integer severity to Canopy's enum. Unknown/None -> INVALID.""" 

63 if value is None: 

64 return AlarmSeverity.INVALID 

65 return _SEVERITY_MAP.get(int(value), AlarmSeverity.INVALID)

97"""EPICS ``menuAlarmStat`` choice names mapped to their alarm status codes. 

98 

99This is the code set Channel Access reports as a value's ``status`` and the one 

100``PVValue.alarm_status`` carries, whatever the protocol. 

101"""

103 

104def alarm_status_from_name(name: object) -> int | None: 

105 """Map an EPICS alarm status name to its ``menuAlarmStat`` code. 

106 

107 PV Access reports the ``menuAlarmStat`` name (``"HIHI"``) in ``alarm.message``, 

108 because its own ``alarm.status`` is a coarser category (DEVICE, DRIVER, RECORD, 

109 ...) rather than the Channel Access code. 

110 

111 Parameters 

112 ---------- 

113 name : object 

114 The status name, matched case-insensitively. The C constant spelling with 

115 an ``_ALARM`` suffix (``"HIHI_ALARM"``) is accepted too. 

116 

117 Returns 

118 ------- 

119 int | None 

120 The ``menuAlarmStat`` code, or ``None`` when ``name`` is not a string naming 

121 a known status, such as a free-text alarm message. 

122 """ 

123 if not isinstance(name, str): 

124 return None 

125 key = name.strip().upper() 

126 if key in ALARM_STATUS_CODES: 

127 return ALARM_STATUS_CODES[key] 

128 return ALARM_STATUS_CODES.get(key.removesuffix("_ALARM"))
