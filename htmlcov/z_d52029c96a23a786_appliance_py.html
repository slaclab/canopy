33def _ns_to_iso(ts_ns: int) -> str: 

34 """Convert UTC nanoseconds to Appliance-friendly ISO-8601 with ms precision and `Z` suffix.""" 

35 dt = datetime.fromtimestamp(ts_ns / 1_000_000_000, tz=UTC) 

36 return dt.isoformat(timespec="milliseconds").replace("+00:00", "Z")

62 """Reachability probe via the Appliance's `getApplianceInfo` management endpoint.""" 

63 start = time.monotonic() 

64 try: 

65 response = await self._client.get("/mgmt/bpl/getApplianceInfo", timeout=2.0) 

66 ok = response.status_code < 500 

67 return AdapterHealth( 

68 healthy=ok, 

69 latency_ms=(time.monotonic() - start) * 1000, 

70 last_error=None if ok else f"HTTP {response.status_code}", 

71 last_check_ts_ns=time.time_ns(), 

72 ) 

73 except Exception as exc: 

74 return AdapterHealth( 

75 healthy=False, 

76 latency_ms=(time.monotonic() - start) * 1000,

78 last_check_ts_ns=time.time_ns(),

81 async def fetch_window( 

82 self, 

83 pv: str, 

84 t0_ns: int, 

85 t1_ns: int, 

86 *, 

87 max_points: int | None = None, # ignored in raw mode; service decimates if needed 

88 ) -> list[Sample]: 

89 """Raw `getData.json` query — one PV, time-window. Returns stored samples. 

90 

91 The Appliance may include the last sample before the window start in 

92 raw responses; we filter to `[t0_ns, t1_ns]` so the contract is 

93 deterministic across Appliance versions. Callers that want the prior 

94 sample use `fetch_last_before`. 

95 """ 

96 params = { 

97 "pv": pv, 

98 "from": _ns_to_iso(t0_ns), 

99 "to": _ns_to_iso(t1_ns), 

100 "donotchunk": "true", 

101 } 

102 response = await self._safe_get("/retrieval/data/getData.json", params=params, pv=pv) 

103 payload = self._parse_json(response) 

104 if not isinstance(payload, list) or not payload: 

105 return [] 

106 samples = self._parse_raw_block(payload[0]) 

107 return [s for s in samples if t0_ns <= s.ts_ns <= t1_ns]

109 async def fetch_last_before(self, pv: str, ts_ns: int) -> Sample | None: 

110 """Most recent stored sample at or before `ts_ns`, or `None`. 

111 

112 A zero-width `getData.json` query (`from == to`) makes the Appliance 

113 return the last known sample at or before that instant. `_ns_to_iso` 

114 floors to millisecond precision, so a sample landing within the final 

115 millisecond before `ts_ns` may be missed — negligible for archive 

116 lookback. 

117 """ 

118 at = _ns_to_iso(ts_ns) 

119 params = { 

120 "pv": pv, 

121 "from": at, 

122 "to": at, 

123 "donotchunk": "true", 

124 } 

125 response = await self._safe_get("/retrieval/data/getData.json", params=params, pv=pv) 

126 payload = self._parse_json(response) 

127 if not isinstance(payload, list) or not payload: 

128 return None 

129 candidates = [s for s in self._parse_raw_block(payload[0]) if s.ts_ns <= ts_ns] 

130 return candidates[-1] if candidates else None

132 async def fetch_binned( 

133 self, 

134 pv: str, 

135 t0_ns: int, 

136 t1_ns: int, 

137 *, 

138 binning: Literal["auto", "time"], 

139 target_buckets: int | None, 

140 bin_size_ns: int | None, 

141 ) -> list[BinnedPoint]: 

142 """Map intent verbs to the Appliance's `optimized_<N>(PV)` operator. 

143 

144 - ``binning="auto"``: ``N = target_buckets``. 

145 - ``binning="time"``: ``N`` is computed so each bucket approximates 

146 ``bin_size_ns``. The Appliance's optimized operator is adaptive, 

147 so widths are approximate; per-bucket `bin_ns` is computed from 

148 the gap between consecutive returned timestamps. 

149 """ 

150 if binning == "auto": 

151 if target_buckets is None or target_buckets < 1: 

152 raise BinningNotSupported("binning='auto' requires target_buckets >= 1") 

153 n = int(target_buckets) 

154 elif binning == "time": 

155 if bin_size_ns is None or bin_size_ns < 1: 

156 raise BinningNotSupported("binning='time' requires bin_size_ns >= 1") 

157 n = max((t1_ns - t0_ns) // bin_size_ns, 1) 

158 else: 

159 raise BinningNotSupported(f"Unsupported binning mode: {binning!r}") 

160 

161 operator_pv = f"optimized_{n}({pv})" 

162 params = { 

163 "pv": operator_pv, 

164 "from": _ns_to_iso(t0_ns), 

165 "to": _ns_to_iso(t1_ns), 

166 "donotchunk": "true", 

167 } 

168 response = await self._safe_get("/retrieval/data/getData.json", params=params, pv=pv) 

169 payload = self._parse_json(response) 

170 if not isinstance(payload, list) or not payload: 

171 return [] 

172 return self._parse_binned_block(payload[0])

174 async def metadata(self, pv: str) -> PVMetadata: 

175 """Pull `EGU` and `PREC` from the Appliance's `meta` block. 

176 

177 The Appliance carries metadata in every `getData.json` response. We 

178 query a small recent window — the `meta` block is populated even when 

179 the time range has no samples, so this works for newly-archived PVs. 

180 Other metadata fields (description, alarm/display limits) are not 

181 carried by the Appliance and stay `None` here; they belong to Hare / 

182 EPICS Connector. 

183 """ 

184 # 1-hour window ending at "now". large enough to land within retention, small enough to keep the response trivial. 

185 now = time.time_ns() 

186 one_hour_ago = now - 3600 * 1_000_000_000 

187 params = { 

188 "pv": pv, 

189 "from": _ns_to_iso(one_hour_ago), 

190 "to": _ns_to_iso(now), 

191 "donotchunk": "true", 

192 } 

193 response = await self._safe_get("/retrieval/data/getData.json", params=params, pv=pv) 

194 payload = self._parse_json(response) 

195 if not isinstance(payload, list) or not payload: 

196 raise PVNotFoundInArchiver(f"No metadata for PV '{pv}' in archiver '{self.name}'") 

197 meta = payload[0].get("meta") or {} 

198 egu = meta.get("EGU") 

199 prec_raw = meta.get("PREC") 

200 try: 

201 precision = int(prec_raw) if prec_raw not in (None, "") else None 

202 except (TypeError, ValueError): 

203 precision = None 

204 return PVMetadata( 

205 pv=pv, 

206 archiver=self.name, 

207 units=egu, 

208 precision=precision, 

209 )

211 async def search(self, pattern: str, *, limit: int) -> list[str]: 

212 """`/retrieval/bpl/searchForPVsRegex` returns whitespace-separated PV names as plain text.""" 

213 try: 

214 response = await self._client.get("/retrieval/bpl/searchForPVsRegex", params={"regex": pattern}) 

215 except httpx.TimeoutException as exc: 

216 raise ArchiverTimeout(f"Appliance '{self.name}' search timed out: {exc}") from exc 

217 except httpx.HTTPError as exc: 

218 raise ArchiverUnavailable(f"Appliance '{self.name}' search unreachable: {exc}") from exc 

219 if not response.is_success: 

220 sc = response.status_code 

221 if sc >= 500: 

222 raise ArchiverUnavailable(f"Appliance '{self.name}' search returned HTTP {sc}") 

223 raise ArchiverBadResponse(f"Appliance '{self.name}' search returned HTTP {sc}") 

224 # Plain text, whitespace-separated; .split() collapses runs of whitespace. 

225 return (response.text or "").split()[:limit]

229 async def _safe_get(self, path: str, *, params: dict, pv: str) -> httpx.Response: 

230 try: 

231 response = await self._client.get(path, params=params) 

232 except httpx.TimeoutException as exc: 

233 raise ArchiverTimeout(f"Appliance {self.name} timed out: {exc}") from exc 

234 except httpx.HTTPError as exc: 

235 raise ArchiverUnavailable(f"Appliance {self.name} unreachable: {exc}") from exc 

236 self._check_status(response, pv) 

237 return response

239 def _check_status(self, response: httpx.Response, pv: str) -> None: 

240 if response.is_success: 

241 return 

242 sc = response.status_code 

243 if sc in (400, 404): 

244 # The Appliance returns 400 for unknown PVs in most deployments. 

245 raise PVNotFoundInArchiver(f"PV '{pv}' not in archiver '{self.name}' (HTTP {sc})") 

246 if sc >= 500: 

247 raise ArchiverUnavailable(f"Appliance '{self.name}' returned HTTP {sc}") 

248 raise ArchiverBadResponse(f"Appliance '{self.name}' returned unexpected HTTP {sc}")

256 def _parse_binned_block(self, block: dict) -> list[BinnedPoint]: 

257 rows = block.get("data") 

258 if not isinstance(rows, list): 

259 raise ArchiverBadResponse(f"Appliance '{self.name}': missing 'data' array in binned response") 

260 points: list[BinnedPoint] = [] 

261 for i, row in enumerate(rows): 

262 try: 

263 secs = int(row["secs"]) 

264 nanos = int(row.get("nanos", 0)) 

265 val = row["val"] 

266 except (KeyError, TypeError, ValueError) as exc: 

267 raise ArchiverBadResponse(f"Appliance '{self.name}': malformed binned row: {exc}") from exc 

268 ts_ns = secs * 1_000_000_000 + nanos 

269 if isinstance(val, int | float): 

270 # Appliance fallback: when the requested window has too few samples 

271 # to bin, optimized_<N> returns the raw sample value as a scalar. 

272 # Promote to a degenerate single-sample bin so callers see a 

273 # uniform BinnedPoint stream. 

274 v = float(val) 

275 mean = mn = mx = v 

276 std = 0.0 

277 cnt = 1 

278 elif isinstance(val, list) and len(val) >= 5: 

279 try: 

280 mean, std, mn, mx, cnt = val[:5] 

281 except (ValueError, TypeError) as exc: 

282 raise ArchiverBadResponse(f"Appliance '{self.name}': cannot unpack binned value: {exc}") from exc 

283 else: 

284 raise ArchiverBadResponse( 

285 f"Appliance '{self.name}': expected scalar or 5-element [mean, std, min, max, count] in optimized " 

286 f"response, got {val!r}" 

287 ) 

288 # bin_ns from the gap to the next bucket (last bucket gets 0). 

289 if i + 1 < len(rows): 

290 try: 

291 next_ts = int(rows[i + 1]["secs"]) * 1_000_000_000 + int(rows[i + 1].get("nanos", 0)) 

292 bin_ns = max(next_ts - ts_ns, 1) 

293 except (KeyError, TypeError, ValueError): 

294 bin_ns = 0 

295 else: 

296 bin_ns = 0 

297 points.append( 

298 BinnedPoint( 

299 ts_ns=ts_ns, 

300 bin_ns=bin_ns, 

301 mean=float(mean), 

302 std_dev=float(std), 

303 min=float(mn), 

304 max=float(mx), 

305 count=int(cnt), 

306 # The Appliance's optimized operator does not return per-bucket 

307 # severity: set NO_ALARM 

308 severity=AlarmSeverity.NO_ALARM, 

309 ) 

310 ) 

311 return points

313 def _parse_raw_block(self, block: dict) -> list[Sample]: 

314 rows = block.get("data") 

315 if not isinstance(rows, list): 

316 raise ArchiverBadResponse(f"Appliance '{self.name}': missing 'data' array in response") 

317 samples: list[Sample] = [] 

318 for row in rows: 

319 try: 

320 secs = int(row["secs"]) 

321 nanos = int(row.get("nanos", 0)) 

322 value = row["val"] 

323 severity = row.get("severity") 

324 except (KeyError, TypeError, ValueError) as exc: 

325 raise ArchiverBadResponse(f"Appliance '{self.name}': malformed sample row: {exc}") from exc 

326 samples.append( 

327 Sample( 

328 ts_ns=secs * 1_000_000_000 + nanos, 

329 value=value, 

330 severity=normalize_severity(severity), 

331 source="archive", 

332 ) 

333 ) 

334 return samples
