1"""Adapter for elog-plus (https://github.com/slaclab/elog-plus), SLAC's e-log backend. 

2 

3Ported from react-squirrel-backend, keeping what it learned about elog-plus: 

4 

5- v1 endpoints only. v2's ``NewEntryDTO`` has no ``additionalAuthors`` and 

6 ignores unknown fields, so a v2 post would silently drop them. 

7- Every response is wrapped in an ``ApiResultResponse`` envelope whose 

8 ``payload`` holds the result. 

9- Entry text is stored as HTML (elog-plus parses it with jsoup to find inline 

10 images) and never run through a markdown processor, so we render it here. 

11 elog-plus doesn't clean the HTML it is sent either, so we do that here too. 

12- elog-plus rejects the whole post if any ``additionalAuthors`` entry is not 

13 an LDAP email it knows. 

14- ``eventAt`` is a ``LocalDateTime`` stored as UTC (the server runs in UTC), 

15 so it goes out in UTC with no offset, and ``loggedAt`` comes back the same way. 

16 

17Each instance holds one app's application token, so elog-plus records that 

18app as the author of everything it posts. 

19"""

51_MARKDOWN_EXTENSIONS = ["extra", "sane_lists", "nl2br"] 

52_HEALTH_TIMEOUT_S = 2.0 

53_ERROR_EXCERPT_CHARS = 500 

54_NO_EMAIL = "elog-plus /v1/users/me returned no email for this app's token"

56# What survives in entry text: elog-plus's own sanitizeEntryText list (jsoup's 

57# Safelist.basic() plus h1-h4 and its entry-reference tag), plus the tables and 

58# rules markdown's "extra" emits. Everything else, raw HTML included, is dropped; 

59# <script> and <style> lose their content too. 

60_ALLOWED_TAGS = { 

61 "a", "b", "blockquote", "br", "cite", "code", "dd", "dl", "dt", "em", "i", "li", "ol", "p", "pre", 

62 "q", "small", "span", "strike", "strong", "sub", "sup", "u", "ul", 

63 "h1", "h2", "h3", "h4", 

64 "table", "thead", "tbody", "tr", "th", "td", "hr", 

65 "elog-entry-ref", 

66} # fmt: skip 

67_ALLOWED_ATTRIBUTES = {"a": {"href"}, "blockquote": {"cite"}, "q": {"cite"}, "elog-entry-ref": {"id"}} 

68_ALLOWED_URL_SCHEMES = {"http", "https", "mailto"}

71def render_markdown(md: str) -> str: 

72 """Render ``md`` to HTML and clean it down to what an e-log entry may contain. 

73 

74 elog-plus stores ``text`` as sent and renders it in places (email, other 

75 clients) that don't clean it, so a key with e-log write access must not be 

76 able to store script or event handlers under an app's name. 

77 """ 

78 html = markdown_lib.markdown(md, extensions=_MARKDOWN_EXTENSIONS, output_format="html") 

79 return nh3.clean(html, tags=_ALLOWED_TAGS, attributes=_ALLOWED_ATTRIBUTES, url_schemes=_ALLOWED_URL_SCHEMES)

82def _ssl_verify() -> ssl.SSLContext | bool: 

83 """The ``verify=`` httpx would use with ``trust_env=True``. 

84 

85 The client turns ``trust_env`` off to ignore the process's proxy variables, 

86 which also turns off ``SSL_CERT_FILE`` and ``SSL_CERT_DIR``. Read those two 

87 here so a deployment that trusts its own CA through them still works. 

88 """ 

89 if cafile := os.environ.get("SSL_CERT_FILE"): 

90 return ssl.create_default_context(cafile=cafile) 

91 if capath := os.environ.get("SSL_CERT_DIR"): 

92 return ssl.create_default_context(capath=capath) 

93 return True

96def _unwrap(body: Any) -> Any: 

97 """Strip elog-plus' ``ApiResultResponse`` envelope when present.""" 

98 if isinstance(body, dict) and "payload" in body: 

99 return body["payload"] 

100 return body

103def _to_elog_datetime(value: datetime) -> str: 

104 """elog-plus takes a UTC ``LocalDateTime``: convert an aware value, keep a naive one.""" 

105 if value.tzinfo is not None: 

106 value = value.astimezone(UTC).replace(tzinfo=None) 

107 return value.isoformat(timespec="seconds")

109 

110def _from_elog_datetime(value: Any) -> datetime | None: 

111 if not value: 

112 return None 

113 try: 

114 parsed = datetime.fromisoformat(str(value)) 

115 except ValueError: 

116 return None 

117 return parsed if parsed.tzinfo is not None else parsed.replace(tzinfo=UTC)

120def _email_of(user: Any) -> str: 

121 if not isinstance(user, dict): 

122 return "" 

123 return user.get("mail") or user.get("email") or ""

126def _error_message(response: httpx.Response) -> str | None: 

127 """elog-plus's own ``errorMessage``, which clients show to say why a post was rejected. 

128 

129 Any other body (an ingress or login page, say) is not passed on. 

130 """ 

131 try: 

132 body = response.json() 

133 except ValueError: 

134 return None 

135 if isinstance(body, dict) and body.get("errorMessage"): 

136 return str(body["errorMessage"])[:_ERROR_EXCERPT_CHARS] 

137 return None

140def _segment(value: str) -> str: 

141 """Quote a client-supplied id for use as one path segment of an elog-plus URL. 

142 

143 The app's token is attached to every request, so an id must never be able 

144 to steer the call to a different elog-plus endpoint. 

145 """ 

146 if value in {".", ".."}: 

147 raise BadRequest(f"Invalid e-log id '{value}'") 

148 return quote(value, safe="")

160@register_elog_provider("elog_plus") 

161class ElogPlusAdapter: 

162 """Calls the elog-plus REST API with one app's application token.""" 

163 

164 provider_name: str = "elog_plus" 

165 

166 def __init__( 

167 self, 

168 app: str, 

169 *, 

170 base_url: str, 

171 token: str, 

172 auth_header: str = "x-vouch-idp-accesstoken", 

173 proxy_url: str | None = None, 

174 timeout_s: float = 15.0, 

175 ) -> None: 

176 self.app = app 

177 self._client = httpx.AsyncClient( 

178 base_url=base_url.rstrip("/"), 

179 headers={auth_header: token}, 

180 timeout=timeout_s, 

181 proxy=proxy_url or None, 

182 # Ignore HTTP(S)_PROXY and friends: the e-log route is set by 

183 # CANOPY_ELOG__PROXY_URL alone, not by whatever the process inherited. 

184 trust_env=False, 

185 verify=_ssl_verify(), 

186 ) 

187 # The token's own email, from /v1/users/me. Filters recent entries. 

188 self._email: str | None = None

190 @classmethod 

191 def from_config(cls, app: str, app_config: ElogAppConfig, config: ElogConfig) -> "ElogPlusAdapter": 

192 if not config.elog_plus.base_url: 

193 raise ElogNotConfigured("CANOPY_ELOG__ELOG_PLUS__BASE_URL is not set") 

194 if app_config.token is None or not app_config.token.get_secret_value(): 

195 raise ElogNotConfigured(f"No e-log token is configured for app '{app}'") 

196 return cls( 

197 app, 

198 base_url=config.elog_plus.base_url, 

199 token=app_config.token.get_secret_value(), 

200 auth_header=config.elog_plus.auth_header, 

201 proxy_url=config.proxy_url, 

202 timeout_s=config.timeout_s, 

203 )

209 async def _call(self, method: str, path: str, **kwargs: Any) -> Any: 

210 """Issue one request and return the unwrapped payload, or raise a domain error.""" 

211 try: 

212 response = await self._client.request(method, path, **kwargs) 

213 except httpx.TimeoutException as exc: 

214 logger.warning("elog-plus %s %s timed out for app %s", method, path, self.app) 

215 raise ElogUpstreamTimeout(f"elog-plus {method} {path} timed out") from exc 

216 except httpx.HTTPError as exc: 

217 # Name the error type only, and drop the cause from the traceback: some 

218 # messages quote the request headers, token included (h11's "Illegal 

219 # header value b'<token>\n'"), and this text reaches logs, the client's 

220 # error detail and the health summary. 

221 error = type(exc).__name__ 

222 logger.warning("elog-plus %s %s failed for app %s: %s", method, path, self.app, error) 

223 raise ElogUpstreamError(f"elog-plus is unreachable ({error})") from None 

224 if not response.is_success: 

225 detail = _error_message(response) 

226 logger.warning( 

227 "elog-plus %s %s returned %d for app %s: %s", 

228 method, 

229 path, 

230 response.status_code, 

231 self.app, 

232 detail or response.text.strip()[:_ERROR_EXCERPT_CHARS], 

233 ) 

234 message = f"elog-plus returned {response.status_code}" 

235 if response.is_redirect or response.status_code == 401: 

236 # The Vouch-gated ingress answers an app token with a login redirect. 

237 message += " (is CANOPY_ELOG__ELOG_PLUS__BASE_URL the application-token ingress?)" 

238 raise ElogUpstreamError(f"{message}: {detail}" if detail else message) 

239 try: 

240 return _unwrap(response.json()) 

241 except ValueError as exc: 

242 raise ElogUpstreamError(f"elog-plus {method} {path} returned a non-JSON body") from exc

244 async def _own_email(self) -> str: 

245 """This token's email, looked up once per adapter (one adapter per app token).""" 

246 if self._email is None: 

247 email = _email_of(await self._call("GET", "/v1/users/me")) 

248 if not email: 

249 # Without it the author filter would drop out and list everyone's entries. 

250 raise ElogUpstreamError(_NO_EMAIL) 

251 self._email = email 

252 return self._email

254 @staticmethod 

255 def _entry_id(payload: Any) -> str: 

256 entry_id = (payload.get("id") or payload.get("entryId")) if isinstance(payload, dict) else payload 

257 if not entry_id: 

258 raise ElogUpstreamError("elog-plus returned no entry id") 

259 return str(entry_id)

261 @staticmethod 

262 async def _entry_body(entry: ElogEntryCreateDTO) -> dict[str, Any]: 

263 """The v1 ``EntryNewDTO`` JSON for ``entry``.""" 

264 body: dict[str, Any] = { 

265 "logbooks": list(entry.logbook_ids), 

266 "title": entry.title, 

267 # Off the event loop: every mounted module shares it, and a long body 

268 # takes a while to render and clean. 

269 "text": await asyncio.to_thread(render_markdown, entry.body_markdown), 

270 "tags": list(entry.tag_ids), 

271 } 

272 if entry.attachment_ids: 

273 body["attachments"] = list(entry.attachment_ids) 

274 if entry.additional_authors: 

275 body["additionalAuthors"] = list(entry.additional_authors) 

276 if entry.important: 

277 body["important"] = True 

278 if entry.event_at is not None: 

279 body["eventAt"] = _to_elog_datetime(entry.event_at) 

280 return body

286 async def list_logbooks(self) -> list[ElogLogbookDTO]: 

287 # elog-plus defaults to the logbooks the token can read, plus every publicly 

288 # readable one. Ask for Write (which includes Admin) so callers only see logbooks 

289 # a post can go to. Root tokens get every logbook either way. 

290 items = await self._call("GET", "/v1/logbooks", params={"filterForAuthorizationTypes": "Write"}) or [] 

291 with _parsing("logbooks"): 

292 return [ElogLogbookDTO(id=item["id"], name=item["name"]) for item in items]

294 async def list_tags(self, logbook_id: str) -> list[ElogTagDTO]:

296 with _parsing("tags"): 

297 return [ElogTagDTO(id=item["id"], name=item["name"]) for item in items]

299 async def search_users(self, search: str, limit: int) -> list[ElogUserDTO]: 

300 items = await self._call("GET", "/v1/users", params={"search": search, "limit": limit}) or [] 

301 with _parsing("users"): 

302 return [ 

303 ElogUserDTO(

305 common_name=item.get("commonName", item.get("name", "")), 

306 surname=item.get("surname", ""), 

307 gecos=item.get("gecos", ""), 

308 mail=item.get("mail", item.get("email", "")), 

309 ) 

310 for item in items

313 async def list_recent_entries(self, logbook_ids: list[str], limit: int) -> list[ElogRecentEntryDTO]: 

314 email = await self._own_email() 

315 params = [("logbooks", logbook_id) for logbook_id in logbook_ids] 

316 params += [("authors", email), ("limit", str(limit))] 

317 items = await self._call("GET", "/v1/entries", params=params) or [] 

318 with _parsing("entries"): 

319 return [ 

320 ElogRecentEntryDTO( 

321 id=item["id"], 

322 title=item.get("title", ""), 

323 logged_at=_from_elog_datetime(item.get("loggedAt") or item.get("eventAt")), 

324 logged_by=item.get("loggedBy", ""), 

325 ) 

326 for item in items 

327 ]

346 async def health(self) -> AdapterHealth: 

347 """Probe with ``GET /v1/users/me``: proves both reachability and that the token is accepted.""" 

348 start = time.monotonic() 

349 try: 

350 email = _email_of(await self._call("GET", "/v1/users/me", timeout=_HEALTH_TIMEOUT_S)) 

351 if not email: 

352 # Recent entries would fail on the same payload, so don't report ok. 

353 raise ElogUpstreamError(_NO_EMAIL) 

354 except Exception as exc: 

355 return AdapterHealth(healthy=False, latency_ms=(time.monotonic() - start) * 1000, last_error=str(exc)) 

356 self._email = email 

357 return AdapterHealth(healthy=True, identity=email, latency_ms=(time.monotonic() - start) * 1000)
