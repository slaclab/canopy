1"""Squirrel module configuration. 

2 

3Sourced from env vars with prefix ``CANOPY_SQUIRREL__`` (nested delimiter ``__``), 

4mirroring the other Canopy modules. A fresh instance per ``get_squirrel_config()`` 

5call (no caching) so tests can vary the environment freely. 

6""" 

7 

8from __future__ import annotations 

9 

10from typing import Self 

11 

12from pydantic import Field, BaseModel, model_validator 

13from pydantic_settings import BaseSettings, SettingsConfigDict 

14 

15 

16class PVCacheConfig(BaseModel): 

17 """The worker-side PV cache that ``POST /snapshots?useCache=true`` reads from. 

18 

19 When enabled, one ``canopy-worker`` (elected through a Redis lease) monitors every 

20 registered setpoint and readback address and keeps its latest value in Redis. 

21 Snapshot create can then read the cache instead of issuing a live read per PV; 

22 anything the cache can't vouch for is still read live. 

23 

24 Cached values are monitor-resolution: alarm changes are posted at once, but a 

25 value change within the record's monitor deadband (MDEL) isn't, so a cached value 

26 can trail a live read by up to that deadband, with its own (older) timestamp. 

27 """ 

28 

29 enabled: bool = False 

30 # How often the leader re-reads the registered PVs from the database and 

31 # subscribes to new addresses / drops removed ones. 

32 reconcile_interval_s: float = Field(60.0, gt=0) 

33 # How often buffered monitor updates are written to Redis (and the heartbeat bumped). 

34 flush_interval_s: float = Field(1.0, gt=0) 

35 # A cache whose heartbeat is older than this is ignored entirely by snapshot create. 

36 heartbeat_max_age_s: float = Field(30.0, gt=0) 

37 # Leader lease length; the leader renews it every third of this. 

38 lease_ttl_s: float = Field(30.0, gt=0) 

39 # Subscriptions opened concurrently while (re)building the monitor set. 

40 subscribe_concurrency: int = Field(200, ge=1) 

41 

42 @model_validator(mode="after") 

43 def _heartbeat_outlasts_a_flush(self) -> Self: 

44 # Otherwise every heartbeat is stale by the next flush and snapshots never use the cache. 

45 if self.heartbeat_max_age_s <= self.flush_interval_s: 

46 raise ValueError( 

47 f"heartbeat_max_age_s ({self.heartbeat_max_age_s:g}) must be longer than " 

48 f"flush_interval_s ({self.flush_interval_s:g})" 

49 ) 

50 return self

63 restore_hooks: str = Field(

76 )

78 # EPICS batch I/O for snapshot create (reads) / restore (writes) over the 

79 # canopy.epics connector: in-flight concurrency, progress-chunk size, per-op 

80 # timeout in seconds (None = connector default / no timeout). 

81 epics_concurrency: int = 64 

82 epics_chunk_size: int = 500 

83 epics_timeout_s: float | None = None

88 @property 

89 def restore_hooks_list(self) -> list[str]: 

90 """Split ``restore_hooks`` into provider keys. 

91 

92 Returns 

93 ------- 

94 list[str] 

95 The comma-separated entries, stripped of surrounding whitespace, with 

96 empty entries dropped and configured order preserved. 

97 """ 

98 return [name.strip() for name in self.restore_hooks.split(",") if name.strip()]
