24class _WSValidationError(ValueError): 

25 """A client frame field was the wrong type. Carries a client-safe reason.""" 

26 

27 

28def _coerce_int(value: object, field: str, default: int = 0) -> int: 

29 """Coerce a JSON numeric to int, or raise. ``None`` yields ``default``. 

30 

31 Rejects strings/bools/objects up front so a value like ``"fast"`` never 

32 reaches the subscription manager (where it used to raise mid-setup and 

33 orphan a shared eval). 

34 """ 

35 if value is None: 

36 return default 

37 if isinstance(value, bool) or not isinstance(value, int | float): 

38 raise _WSValidationError(f"'{field}' must be a number") 

39 return int(value) 

40 

41 

42def _coerce_opt_int(value: object, field: str) -> int | None: 

43 """Like :func:`_coerce_int` but preserves ``None`` (meaning "unchanged").""" 

44 return None if value is None else _coerce_int(value, field)

47async def _send_error(send_fn, code: str, message: str, *, sub_id: str | None = None) -> None: 

48 frame: dict = {"type": "error", "code": code, "message": message} 

49 if sub_id is not None: 

50 frame["id"] = sub_id 

51 await send_fn(frame)

54@router.websocket("/formulas", dependencies=[Security(ws_require_read_access)]) 

55async def ws_formulas(websocket: WebSocket, manager: SubscriptionManagerDep) -> None: 

56 """Live formula evaluation stream. 

57 

58 Client -> server: 

59 - ``{"type": "subscribe", "spec": {...}, "t0Ns": ..., "t1Ns": ...}`` 

60 - ``{"type": "unsubscribe", "id": "<server-assigned>"}`` 

61 - ``{"type": "ping"}`` 

62 

63 Server -> client: 

64 - ``{"type": "subscribed", "id": "<assigned>"}`` 

65 - ``{"type": "result", "id": "...", "data": {...}}`` 

66 - ``{"type": "error", "code": "...", "message": "..."}`` 

67 - ``{"type": "pong", "timestamp": T}`` 

68 """ 

69 await websocket.accept() 

70 client_id = uuid.uuid4().hex 

71 send_lock = asyncio.Lock() 

72 

73 async def send_fn(msg: dict) -> None: 

74 async with send_lock: 

75 await websocket.send_json(msg) 

76 

77 try: 

78 while True: 

79 # A malformed frame sends an error and keeps the socket open -- one 

80 # bad message must not drop the connection and all its subscriptions. 

81 try: 

82 message = await websocket.receive_json() 

83 except WebSocketDisconnect: 

84 raise 

85 except Exception: 

86 await _send_error(send_fn, ErrorCode.INVALID_MESSAGE.value, "message must be valid JSON") 

87 continue 

88 if not isinstance(message, dict): 

89 await _send_error(send_fn, ErrorCode.INVALID_MESSAGE.value, "message must be a JSON object") 

90 continue 

91 mtype = message.get("type") 

92 

93 if mtype == "subscribe": 

94 spec_raw = message.get("spec") 

95 

96 if not spec_raw: 

97 await _send_error(send_fn, ErrorCode.INVALID_MESSAGE.value, "subscribe requires 'spec'") 

98 continue 

99 

100 try: 

101 t0_ns = _coerce_int(message.get("t0Ns"), "t0Ns") 

102 t1_ns = _coerce_int(message.get("t1Ns"), "t1Ns") 

103 interval_ms = _coerce_int(message.get("intervalMs"), "intervalMs") 

104 except _WSValidationError as exc: 

105 await _send_error(send_fn, ErrorCode.INVALID_MESSAGE.value, str(exc)) 

106 continue 

107 

108 try: 

109 spec = FormulaSpec(**spec_raw) 

110 except Exception as exc: 

111 await _send_error(send_fn, ErrorCode.INVALID_SPEC.value, f"invalid spec: {exc}") 

112 continue 

113 

114 sub_id = uuid.uuid4().hex 

115 await send_fn({"type": "subscribed", "id": sub_id}) 

116 try: 

117 await manager.add_subscription( 

118 client_id, sub_id, spec, t0_ns, t1_ns, send_fn, interval_ms=interval_ms 

119 ) 

120 except CanopyError as exc: 

121 await _send_error(send_fn, exc.error_code.value, exc.detail, sub_id=sub_id) 

122 except Exception as exc: 

123 logger.warning("Fox subscribe failed for %s: %s", sub_id, exc) 

124 await _send_error(send_fn, ErrorCode.UNHANDLED_EXCEPTION.value, str(exc), sub_id=sub_id) 

125 

126 elif mtype == "update": 

127 sub_id = message.get("id", "") 

128 try: 

129 t0_update = _coerce_opt_int(message.get("t0Ns"), "t0Ns") 

130 interval_update = _coerce_opt_int(message.get("intervalMs"), "intervalMs") 

131 except _WSValidationError as exc: 

132 await _send_error(send_fn, ErrorCode.INVALID_MESSAGE.value, str(exc)) 

133 continue 

134 spec_update = message.get("spec") 

135 

136 if not sub_id: 

137 await _send_error(send_fn, ErrorCode.INVALID_MESSAGE.value, "update requires 'id'") 

138 continue 

139 

140 parsed_spec = None 

141 if spec_update is not None: 

142 try: 

143 parsed_spec = FormulaSpec(**spec_update) 

144 except Exception as exc: 

145 await _send_error(send_fn, ErrorCode.INVALID_SPEC.value, f"invalid spec: {exc}", sub_id=sub_id) 

146 continue 

147 

148 # Guard like the subscribe handler: a failure while starting the 

149 # new eval must send an error frame, not fall through to the outer 

150 # handler and drop the whole connection (and all its subscriptions). 

151 try: 

152 ok = await manager.update_subscription( 

153 client_id, sub_id, send_fn, spec=parsed_spec, t0_ns=t0_update, interval_ms=interval_update 

154 ) 

155 except CanopyError as exc: 

156 await _send_error(send_fn, exc.error_code.value, exc.detail, sub_id=sub_id) 

157 continue 

158 except Exception as exc: 

159 logger.warning("Fox update failed for %s: %s", sub_id, exc) 

160 await _send_error(send_fn, ErrorCode.UNHANDLED_EXCEPTION.value, str(exc), sub_id=sub_id) 

161 continue 

162 

163 if ok: 

164 await send_fn({"type": "updated", "id": sub_id}) 

165 else: 

166 await _send_error(send_fn, ErrorCode.NOT_FOUND.value, "subscription not found", sub_id=sub_id) 

167 

168 elif mtype == "unsubscribe": 

169 sub_id = message.get("id", "") 

170 await manager.remove_subscription(client_id, sub_id) 

171 

172 elif mtype == "ping": 

173 await send_fn({"type": "pong", "timestamp": time.time()}) 

174 

175 else: 

176 await _send_error(send_fn, ErrorCode.INVALID_MESSAGE.value, f"unknown message type: {mtype!r}") 

177 except WebSocketDisconnect: 

178 await manager.remove_client(client_id) 

179 except Exception as exc: 

180 logger.warning("Fox WS error for client %s: %s", client_id, exc) 

181 await manager.remove_client(client_id)
