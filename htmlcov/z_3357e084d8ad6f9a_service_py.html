44def _normalize_modules(modules: list[str]) -> list[str]: 

45 """Validate and canonicalize a requested module list. 

46 

47 Callers must pass ``[WILDCARD]`` explicitly for "every module" — there is no 

48 implicit default here. An empty list is rejected the same way an 

49 all-scopes-false key is: it isn't a credential, it's a mistake that will be 

50 diagnosed as an outage. 

51 """ 

52 cleaned = {m.strip() for m in modules if m.strip()} 

53 if not cleaned: 

54 raise BadRequest(f"Cannot create an API key scoped to no modules; pass at least one, or '{WILDCARD}' for all") 

55 known, unknown = split_modules(cleaned) 

56 if unknown: 

57 raise BadRequest(f"Unknown module(s) {sorted(unknown)}; valid modules are {sorted(MODULES)}") 

58 return sorted(known)

61def _require_module_containment(target_modules, acting_modules) -> None: 

62 """Refuse an admin action that reaches outside the acting key's own modules. 

63 

64 ``acting_modules`` is ``None`` for the CLI scripts, which mint/deactivate 

65 straight against the database with no acting key to contain against — 

66 that path stays unrestricted, same as before this check existed. Over 

67 HTTP an admin key is only ever a delegated admin for its own modules 

68 unless it holds ``WILDCARD``, so ``ops:admin@core`` cannot mint or 

69 deactivate a key scoped to ``squirrel``. 

70 """ 

71 if acting_modules is None or WILDCARD in acting_modules: 

72 return 

73 if outside := set(target_modules) - set(acting_modules): 

74 raise InsufficientModuleAccess(f"Acting API key is not scoped to module(s) {sorted(outside)}")

84 async def create_key( 

85 self, 

86 app_name: str, 

87 *, 

88 read: bool = False, 

89 write: bool = False, 

90 admin: bool = False, 

91 modules: list[str], 

92 acting_key_modules: list[str] | None = None, 

93 ) -> ApiKeyCreateResultDTO: 

94 """Mint a key and return it with its plaintext token — the only time the 

95 token exists outside the caller's hands. Only its hash is stored. 

96 

97 An existing active key for the same ``app_name`` is *not* an error: that 

98 overlap is how a rotation works. 

99 

100 ``modules`` restricts which mounted sub-apps (or ``core``, for key 

101 management itself) the key may call. Pass ``[WILDCARD]`` for every module 

102 — callers must choose explicitly, there is no implicit default here. 

103 

104 ``acting_key_modules`` is the module set of the admin key making this 

105 request (``None`` for the CLI). A module-scoped admin can only delegate 

106 modules it itself holds — it cannot mint a key scoped wider than itself. 

107 """ 

108 app_name = app_name.strip() 

109 if not app_name: 

110 raise BadRequest("app_name must not be empty") 

111 if not (read or write or admin): 

112 raise BadRequest("Cannot create an API key with no access; pass at least one scope") 

113 module_list = _normalize_modules(modules) 

114 _require_module_containment(module_list, acting_key_modules) 

115 

116 token = mint_token() 

117 created = await self.repo.create( 

118 ApiKey( 

119 app_name=app_name, 

120 token_hash=hash_token(token), 

121 read_access=read, 

122 write_access=write, 

123 admin_access=admin, 

124 modules=module_list, 

125 ) 

126 ) 

127 # The token itself is never logged — app_name and id are the audit trail. 

128 logger.info("Created API key id=%s app_name=%s", created.id, created.app_name) 

129 return ApiKeyCreateResultDTO(token=token, **to_dto(created).model_dump())

132 if app_name is not None:

138 async def _find(self, key_id: str) -> ApiKey | None: 

139 """Look a key up by id, treating a malformed id as simply absent. 

140 

141 The id column is a UUID, so handing asyncpg a mistyped id raises a DataError 

142 — a 500 through a route, and a wall of traceback from the CLI. Validate first 

143 and report it the way a wrong-but-well-formed id is reported. 

144 """ 

145 try: 

146 UUID(key_id) 

147 except ValueError: 

148 return None 

149 return await self.repo.get_by_id(key_id) 

150 

151 async def get_by_id(self, key_id: str) -> ApiKeyDTO | None: 

152 api_key = await self._find(key_id) 

153 return to_dto(api_key) if api_key else None

155 async def count(self, active_only: bool = False) -> int: 

156 return await self.repo.count(active_only)

158 async def deactivate_key( 

159 self, 

160 key_id: str, 

161 *, 

162 acting_key_id: str | None = None, 

163 acting_key_modules: list[str] | None = None, 

164 force: bool = False, 

165 ) -> ApiKeyDTO: 

166 """Soft-delete a key. Idempotent-adjacent: deactivating an already inactive 

167 key is a no-op rather than an error, so a retried revocation still succeeds. 

168 

169 ``acting_key_id`` is the id of the key that authenticated this call (``None`` 

170 for the CLI, which has no HTTP request to guard). Deactivating the key you're 

171 making the request with would lock an operator out mid-session, so it's 

172 refused unless ``force`` is set. 

173 

174 ``acting_key_modules`` (``None`` for the CLI) contains the same delegation 

175 check as ``create_key``: a module-scoped admin cannot deactivate a key 

176 scoped to a module it does not itself hold. 

177 """ 

178 api_key = await self._find(key_id) 

179 if api_key is None: 

180 raise ApiKeyNotFound(f"No API key with id '{key_id}'") 

181 _require_module_containment(api_key.modules, acting_key_modules) 

182 if not force and acting_key_id is not None and api_key.id == acting_key_id: 

183 raise ApiKeySelfDeactivation( 

184 "Refusing to deactivate the API key authenticating this request; pass force=true to override" 

185 ) 

186 if not api_key.is_active: 

187 return to_dto(api_key) 

188 deactivated = await self.repo.deactivate(api_key) 

189 invalidate_key_id(api_key.id) 

190 logger.info("Deactivated API key id=%s app_name=%s", api_key.id, api_key.app_name) 

191 return to_dto(deactivated)
