47def _normalize_modules(modules: list[str]) -> list[str]: 

48 """Validate and canonicalize a requested module list. 

49 

50 Callers must pass ``[WILDCARD]`` explicitly for "every module" — there is no 

51 implicit default here. An empty list is rejected the same way an 

52 all-scopes-false key is: it isn't a credential, it's a mistake that will be 

53 diagnosed as an outage. 

54 """ 

55 cleaned = {m.strip() for m in modules if m.strip()} 

56 if not cleaned: 

57 raise BadRequest(f"Cannot create an API key scoped to no modules; pass at least one, or '{WILDCARD}' for all") 

58 known, unknown = split_modules(cleaned) 

59 if unknown: 

60 raise BadRequest(f"Unknown module(s) {sorted(unknown)}; valid modules are {sorted(MODULES)}") 

61 return sorted(known)

64def _require_module_containment(target_modules, acting_modules) -> None: 

65 """Refuse an admin action that reaches outside the acting key's own modules. 

66 

67 ``acting_modules`` is ``None`` for the CLI scripts, which mint/deactivate 

68 straight against the database with no acting key to contain against — 

69 that path stays unrestricted, same as before this check existed. Over 

70 HTTP an admin key is only ever a delegated admin for its own modules 

71 unless it holds ``WILDCARD``, so ``ops:admin@core`` cannot mint or 

72 deactivate a key scoped to ``squirrel``. 

73 """ 

74 if acting_modules is None or WILDCARD in acting_modules: 

75 return 

76 if outside := set(target_modules) - set(acting_modules): 

77 raise InsufficientModuleAccess(f"Acting API key is not scoped to module(s) {sorted(outside)}")

80async def _require_facility_containment(target_facilities: list[str], acting_key_facilities: list[str] | None) -> None: 

81 """Refuse an admin action that reaches outside the acting key's own facility grant. 

82 

83 Mirrors ``_require_module_containment``, but facility containment needs hierarchy 

84 expansion — a grant on a parent covers a child the target list names directly — 

85 so each target slug is checked by resolving its descendant set and testing it 

86 against the acting key's own resolved scope, rather than comparing slug sets 

87 directly. ``acting_key_facilities`` is ``None`` for the CLI, which stays 

88 unrestricted, same as module containment. 

89 

90 An acting key not holding the wildcard can never delegate the wildcard: there is 

91 no finite resolved scope to compare it against, since a future facility would 

92 fall inside it too. 

93 """ 

94 if acting_key_facilities is None or WILDCARD in acting_key_facilities: 

95 return 

96 if WILDCARD in target_facilities: 

97 raise InsufficientFacilityAccess("Acting API key is not scoped to every facility and cannot delegate '*'") 

98 acting_scope = await facilities_gateway.resolve_key_facility_scope(frozenset(acting_key_facilities)) 

99 outside = set() 

100 for slug in target_facilities: 

101 slug_scope = await facilities_gateway.resolve_key_facility_scope(frozenset({slug})) 

102 if slug_scope and not slug_scope <= acting_scope: 

103 outside.add(slug) 

104 if outside: 

105 raise InsufficientFacilityAccess(f"Acting API key is not scoped to facility/facilities {sorted(outside)}")

108def _normalize_facilities(facilities: list[str]) -> list[str]: 

109 """Canonicalize a requested facility list. 

110 

111 Unlike ``_normalize_modules``, an empty list is valid and meaningful — 

112 "provisioned but not yet assigned to a facility" / "data-suspended without 

113 revoking the credential" — so it is returned as-is rather than rejected. 

114 

115 Whether the named slugs *exist* is checked separately, in ``create_key``, 

116 where there's a session in hand; this stays a pure string transform. 

117 """ 

118 cleaned = {f.strip() for f in facilities if f.strip()} 

119 if WILDCARD in cleaned: 

120 return [WILDCARD] 

121 return sorted(cleaned)

131 async def create_key( 

132 self, 

133 app_name: str, 

134 *, 

135 read: bool = False, 

136 write: bool = False, 

137 admin: bool = False, 

138 modules: list[str], 

139 acting_key_modules: list[str] | None = None, 

140 facilities: list[str] | None = None, 

141 acting_key_facilities: list[str] | None = None, 

142 ) -> ApiKeyCreateResultDTO: 

143 """Mint a key and return it with its plaintext token — the only time the 

144 token exists outside the caller's hands. Only its hash is stored. 

145 

146 An existing active key for the same ``app_name`` is *not* an error: that 

147 overlap is how a rotation works. 

148 

149 ``modules`` restricts which mounted sub-apps (or ``core``, for key 

150 management itself) the key may call. Pass ``[WILDCARD]`` for every module 

151 — callers must choose explicitly, there is no implicit default here. 

152 

153 ``acting_key_modules`` is the module set of the admin key making this 

154 request (``None`` for the CLI). A module-scoped admin can only delegate 

155 modules it itself holds — it cannot mint a key scoped wider than itself. 

156 

157 ``facilities`` defaults to no grants (``[]``) rather than every module's 

158 implicit ``[WILDCARD]`` — unlike modules, an unassigned key is a valid, 

159 unprivileged state here, not a mistake. 

160 

161 ``acting_key_facilities`` (``None`` for the CLI) is the same delegation check 

162 as ``acting_key_modules``, for facility scope: a facility-scoped admin cannot 

163 mint a key granted a facility outside its own resolved scope, or the wildcard. 

164 """ 

165 app_name = app_name.strip() 

166 if not app_name: 

167 raise BadRequest("app_name must not be empty") 

168 if not (read or write or admin): 

169 raise BadRequest("Cannot create an API key with no access; pass at least one scope") 

170 module_list = _normalize_modules(modules) 

171 _require_module_containment(module_list, acting_key_modules) 

172 facility_list = _normalize_facilities(facilities or []) 

173 # A grant naming a nonexistent facility is a hard reject: it resolves to 

174 # "matches nothing" at request time, so the key would authenticate and then 

175 # see and write nothing, with no error to point at the typo. `[WILDCARD]` 

176 # names no slug, so there's nothing to check. 

177 if facility_list and facility_list != [WILDCARD]: 

178 unknown = set(facility_list) - await facilities_gateway.known_slugs(facility_list) 

179 if unknown: 

180 raise BadRequest( 

181 f"Unknown facility slug(s) {sorted(unknown)}; create them first " 

182 f"(scripts/create_facility.py or POST /v1/facilities)" 

183 ) 

184 await _require_facility_containment(facility_list, acting_key_facilities) 

185 

186 token = mint_token() 

187 created = await self.repo.create( 

188 ApiKey( 

189 app_name=app_name, 

190 token_hash=hash_token(token), 

191 read_access=read, 

192 write_access=write, 

193 admin_access=admin, 

194 modules=module_list, 

195 facilities=facility_list, 

196 ) 

197 ) 

198 # The token itself is never logged — app_name and id are the audit trail. 

199 logger.info("Created API key id=%s app_name=%s", created.id, created.app_name) 

200 return ApiKeyCreateResultDTO(token=token, **to_dto(created).model_dump())

209 async def _find(self, key_id: str) -> ApiKey | None: 

210 """Look a key up by id, treating a malformed id as simply absent. 

211 

212 The id column is a UUID, so handing asyncpg a mistyped id raises a DataError 

213 — a 500 through a route, and a wall of traceback from the CLI. Validate first 

214 and report it the way a wrong-but-well-formed id is reported. 

215 """ 

216 try: 

217 UUID(key_id) 

218 except ValueError: 

219 return None 

220 return await self.repo.get_by_id(key_id)

229 async def deactivate_key( 

230 self, 

231 key_id: str, 

232 *, 

233 acting_key_id: str | None = None, 

234 acting_key_modules: list[str] | None = None, 

235 force: bool = False, 

236 acting_key_facilities: list[str] | None = None, 

237 ) -> ApiKeyDTO: 

238 """Soft-delete a key. Idempotent-adjacent: deactivating an already inactive 

239 key is a no-op rather than an error, so a retried revocation still succeeds. 

240 

241 ``acting_key_id`` is the id of the key that authenticated this call (``None`` 

242 for the CLI, which has no HTTP request to guard). Deactivating the key you're 

243 making the request with would lock an operator out mid-session, so it's 

244 refused unless ``force`` is set. 

245 

246 ``acting_key_modules`` (``None`` for the CLI) contains the same delegation 

247 check as ``create_key``: a module-scoped admin cannot deactivate a key 

248 scoped to a module it does not itself hold. ``acting_key_facilities`` is the 

249 same check for facility scope. 

250 """ 

251 api_key = await self._find(key_id) 

252 if api_key is None: 

253 raise ApiKeyNotFound(f"No API key with id '{key_id}'") 

254 _require_module_containment(api_key.modules, acting_key_modules) 

255 await _require_facility_containment(api_key.facilities, acting_key_facilities) 

256 if not force and acting_key_id is not None and api_key.id == acting_key_id: 

257 raise ApiKeySelfDeactivation( 

258 "Refusing to deactivate the API key authenticating this request; pass force=true to override" 

259 ) 

260 if not api_key.is_active: 

261 return to_dto(api_key) 

262 deactivated = await self.repo.deactivate(api_key) 

263 invalidate_key_id(api_key.id) 

264 logger.info("Deactivated API key id=%s app_name=%s", api_key.id, api_key.app_name) 

265 return to_dto(deactivated)
